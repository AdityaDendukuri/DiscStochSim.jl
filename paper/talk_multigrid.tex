\section{The RDME}
% ================================================================

\begin{frame}{The Reaction-Diffusion Master Equation
  \citeref{[Erban et al.\ 2007]}}

  \textbf{Setting.}
  Domain $\Omega \subset \mathbb{R}^d$ partitioned into $K$ voxels of side length $h$.
  $N$ species with diffusion coefficients $D_1,\ldots,D_N$.
  State: $x_{ik}$ = molecules of species $i$ in voxel $k$;
  $\bx_k = (x_{1k},\ldots,x_{Nk})$ = local voxel state.

  \medskip
  \begin{center}
  \begin{tikzpicture}[scale=1.1]
    % voxel boxes
    \foreach \k in {1,2,3,4} {
      \draw[thick,fill=blue!10] (\k-1,0) rectangle (\k,0.9);
      \node[font=\normalsize] at (\k-0.5,0.45) {$\Omega_\k$};
    }
    % clean horizontal diffusion arrows above boxes
    \foreach \k in {1,2,3} {
      \draw[<->,fastblue,very thick] (\k-0.35,1.25) -- (\k+0.35,1.25);
    }
    \node[fastblue,font=\small] at (2,1.65)
      {diffusion: rate $d_i = D_i/h^2$ per molecule};
    % reaction arrow below box
    \draw[->,slowred,thick] (2.5,0) -- (2.5,-0.45);
    \node[slowred,font=\small] at (2.5,-0.68)
      {reactions: propensity $\alpha_r(\bx_k)$ in each voxel};
  \end{tikzpicture}
  \end{center}

  \medskip
  \[
    \frac{dp}{dt} = A_{\mathrm{RDME}}\, p
    \;=\; \Bigl(\sum_k A_{\mathrm{rxn}}^{(k)}
          \;+\; \sum_{\substack{k,k'\\ \text{adjacent}}}
          A_{\mathrm{diff}}^{(kk')}\Bigr)\, p
  \]

\end{frame}

% ----------------------------------------------------------------
\begin{frame}{The Computational Bottleneck: FSP
  \citeref{[Munsky \& Khammash 2006]}}

  The \textbf{Finite State Projection} (FSP) cuts the infinite state space
  $\Znn^{N\times K}$ down to a finite active set $\mathcal{S}$
  and solves the resulting ODE exactly.
  $n_{\max}$ = how many molecules we allow per species per voxel.

  \bigskip
  \textbf{State space size:} $|\mathcal{S}| \sim (n_{\max}+1)^{N \cdot K}$

  \bigskip
  \begin{center}
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{ccc|c}
    \toprule
    $N$ & $K$ & $n_{\max}$ & $|\mathcal{S}|$ \\
    \midrule
    1 & 4  & 12 & $13^4 = 28{,}561$ \quad
      \textcolor{coarsegreen}{tractable} \\
    1 & 8  & 12 & $13^8 \approx 8\times10^8$ \quad
      \textcolor{slowred}{intractable} \\
    1 & 16 & 12 & $13^{16}\approx 7\times10^{17}$ \quad
      \textcolor{slowred}{hopeless} \\
    \bottomrule
  \end{tabular}
  \end{center}

  \bigskip
  \begin{block}{Goal}
    Exploit the \emph{spatial structure} of the RDME to reduce computation
    to a sequence of small, coarse-level FSP solves.
  \end{block}

\end{frame}

\section{Multigrid Method}
% ================================================================

\begin{frame}{Spatial Coarsening via Lumpability
  \citeref{[Kemeny \& Snell 1960]}}

  \textbf{Lumpability (Kemeny \& Snell):}
  if a set of states transition among themselves \emph{very fast},
  you can collapse them into a single macro-state.
  The internal distribution settles to a known stationary form ---
  that's what we use as the prolongation operator.

  \medskip
  \textbf{Idea:} merge pairs of adjacent fine voxels into one coarse voxel.
  Coarse count for species $i$, coarse voxel $j$:
  \[
    \bar{x}_{ij} = x_{i,2j-1} + x_{i,2j}
  \]

  \begin{center}
  \begin{tikzpicture}[scale=0.95]
    \draw[thick,fill=blue!12] (0,1.2) rectangle (1.4,2.0);
    \draw[thick,fill=blue!12] (1.4,1.2) rectangle (2.8,2.0);
    \draw[thick,fill=red!12]  (2.8,1.2) rectangle (4.2,2.0);
    \draw[thick,fill=red!12]  (4.2,1.2) rectangle (5.6,2.0);
    \node at (0.7,1.6) {$k_1$};  \node at (2.1,1.6) {$k_2$};
    \node at (3.5,1.6) {$k_3$};  \node at (4.9,1.6) {$k_4$};
    \node[left,font=\small] at (0,1.6) {fine:};
    \draw[thick,fill=blue!28] (0,0) rectangle (2.8,0.8);
    \draw[thick,fill=red!28]  (2.8,0) rectangle (5.6,0.8);
    \node at (1.4,0.4) {$\bar{k}_1$};  \node at (4.2,0.4) {$\bar{k}_2$};
    \node[left,font=\small] at (0,0.4) {coarse:};
    \draw[->,thick,gray!70] (1.4,1.2) -- (1.4,0.8)
      node[midway,right,font=\footnotesize]{$\calR$: sum counts};
    \draw[->,thick,coarsegreen!70!black] (1.1,0.8) -- (0.7,1.2)
      node[midway,left,font=\footnotesize]{$\calP$: split counts};
  \end{tikzpicture}
  \end{center}

  \textbf{Lumpability parameter} (ratio of inter- to intra-pair diffusion rates):
  \[
    \epsilon = \frac{d_{\mathrm{inter}}}{d_{\mathrm{intra}}}
    = \frac{D/(2h)^2}{D/h^2} = \frac{1}{4}
    \quad \text{per level (uniform grid)}
  \]
  Small $\epsilon \Rightarrow$ coarsening is accurate.

\end{frame}

% ----------------------------------------------------------------
\begin{frame}{Restriction and Prolongation}

  \textbf{Restriction} $\calR$: go from fine to coarse by summing over all
  fine states that share the same coarse counts.
  \[
    (\calR\, p)(\bar{\bx})
    = \sum_{\bx\,:\; x_{i,2j-1}+x_{i,2j}=\bar{x}_{ij}\;\forall i,j} p(\bx)
  \]

  \medskip
  \textbf{Prolongation} $\calP$ \citeref{[Kemeny \& Snell 1960; Simon \& Ando 1961]}:
  go from coarse back to fine.
  With $n = \bar{x}_{ij}$ total molecules in coarse voxel $j$,
  fast intra-voxel diffusion drives the split to stationarity:
  \[
    \pi(m \mid n) = \Bin\!\left(n,\;\tfrac{1}{2}\right),
    \quad m = 0,1,\ldots,n
  \]
  (detailed balance; equal-volume sub-voxels). Then:
  \[
    (\calP\,\bar{p})(\bx)
    = \bar{p}(\bar{\bx})\cdot\prod_{i,j}\pi(x_{i,2j-1}\mid\bar{x}_{ij})
  \]

  \begin{block}{Consistency: $\calR\,\calP = I$}
    If you go coarse then fine, you get back exactly where you started.
  \end{block}

\end{frame}

% ----------------------------------------------------------------
\begin{frame}{Coarse-Level Generator}

  Petrov--Galerkin projection:
  $\bar{A} = \calR\, A_{\mathrm{RDME}}\, \calP$,
  where $A_{\mathrm{RDME}} = A_{\mathrm{rxn}} + A_{\mathrm{intra}} + A_{\mathrm{inter}}$:
  \begin{itemize}\setlength\itemsep{0pt}
    \item $A_{\mathrm{intra}}$: diffusion within each fine pair (fast; rate $D/h^2$)
    \item $A_{\mathrm{inter}}$: diffusion between coarse voxels (slow; rate $D/(2h)^2$)
  \end{itemize}

  \medskip
  \begin{center}
  \renewcommand{\arraystretch}{1.45}
  \begin{tabular}{p{3.6cm}p{7cm}}
    \toprule
    Term & Result \\
    \midrule
    $\calR\, A_{\mathrm{intra}}\, \calP$ &
      $=0$\; ($\Bin$ is stationary for $A_{\mathrm{intra}}$;
      fast intra-voxel diffusion drops out) \\
    $\calR\, A_{\mathrm{inter}}\, \calP$ &
      Coarse diffusion at rate $D/(2h)^2$, same structural form \\
    $\calR\, A_{\mathrm{rxn}}\, \calP$ &
      \textbf{Exact} for linear (0th- and 1st-order) reactions;
      error $O(1/n)$ for nonlinear \\
    \bottomrule
  \end{tabular}
  \end{center}

  \begin{alertblock}{Nice structure}
    $\bar{A}$ is itself an RDME on a coarser grid ---
    so you can keep coarsening recursively all the way down.
  \end{alertblock}

\end{frame}

% ----------------------------------------------------------------
\begin{frame}{The V-Cycle \citeref{[Briggs et al.\ 2000]}}

  Split $A_{\mathrm{RDME}} = A_{\mathrm{fast}} + A_{\mathrm{slow}}$,
  where $A_{\mathrm{fast}} = A_{\mathrm{intra}}$ and
  $A_{\mathrm{slow}} = A_{\mathrm{rxn}} + A_{\mathrm{inter}}$.

  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{block}{V-Cycle: $p(t) \to p(t+\Delta t)$}
        \begin{enumerate}
          \setlength\itemsep{0.35em}
          \item \textbf{Pre-smooth:}
            $\tilde p \gets e^{\tau A_{\mathrm{fast}}} p$ \\
            {\small Run fast diffusion for time $\tau$ so molecules
            even out within each merged pair}
          \item \textbf{Restrict:}
            $p^c \gets \calR\,\tilde p$ \\
            {\small Collapse to the coarse level}
          \item \textbf{Coarse solve (FSP):}
            $\bar p^c \gets e^{\Delta t\,\bar A}\, p^c$ \\
            {\small Take a full time step on the much smaller coarse problem}
          \item \textbf{Prolongate correction:}
            $p \gets \tilde p + \calP(\bar p^c - p^c)$ \\
            {\small Fold the coarse correction back into the fine solution}
          \item \textbf{Post-smooth:}
            $p \gets e^{\tau A_{\mathrm{fast}}} p$
        \end{enumerate}
      \end{block}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{center}
      \begin{tikzpicture}[every node/.style={font=\small}, scale=0.98]
        \node[draw,rounded corners,fill=blue!22,
              minimum width=3.1cm,minimum height=0.62cm]
          (f) at (0,3.4) {level $\ell=0$ (fine)};
        \node[draw,rounded corners,fill=blue!13,
              minimum width=2.7cm,minimum height=0.62cm]
          (m) at (-0.3,2.1) {level $\ell=1$};
        \node[draw,rounded corners,fill=coarsegreen!28,
              minimum width=2.3cm,minimum height=0.62cm]
          (c) at (0,0.8) {level $\ell=L$ (coarsest)};
        \draw[->,thick] (f.south west) -- (m.north)
          node[midway,left,font=\footnotesize]{restrict};
        \draw[->,thick] (m.south) -- (c.north west)
          node[midway,left,font=\footnotesize]{restrict};
        \draw[->,thick,slowred] (c.north east) -- (m.south east)
          node[midway,right,font=\footnotesize]{prolong};
        \draw[->,thick,slowred] (m.north east) -- (f.south east)
          node[midway,right,font=\footnotesize]{prolong};
        \node[coarsegreen!70!black,font=\footnotesize] at (0,0.28)
          {FSP solve};
      \end{tikzpicture}
      \end{center}

      \smallskip
      \textbf{Why this helps:}\\[0.2em]
      The coarse level isn't stiff ---
      you can take steps of size $\Delta t \sim 1/\|A_{\mathrm{slow}}\|$
      instead of the tiny fine-level limit $\sim 1/\|A_{\mathrm{fast}}\|$.\\[0.2em]
      Speedup $\sim d_{\mathrm{intra}} / k_{\mathrm{rxn}} \gg 1$
    \end{column}
  \end{columns}

\end{frame}

\begin{frame}{Adaptive Refinement
  \citeref{[Dendukuri et al.\ 2025]}}

  \textbf{Total inter-voxel flux} between adjacent voxels $k$ and $k'$:
  \[
    \Phi_{kk'} = \sum_{i=1}^N d_i\,\E[x_{ik}]
    \qquad \text{(expected molecules crossing per unit time)}
  \]
  Simple rule: merge if $\Phi_{kk'}$ is small. But this breaks in two ways:

  \begin{enumerate}
    \setlength\itemsep{0.6em}
    \item \textbf{Background inactivity.}
      Almost no molecules around, so $\Phi_{kk'}\approx 0$ no matter what $d_i$ is.
      \textcolor{coarsegreen}{Fine to merge.}

    \item \textbf{Spatial bottleneck.}
      Molecules are there, but $d_i$ is just slow (membrane, narrow channel).
      $\Phi_{kk'}$ looks small, but this pair is a gateway between two regions.
      \textcolor{slowred}{Merging here is wrong.}
  \end{enumerate}

  \medskip
  \textbf{Fix:} refine pair $\{k,k'\}$ if either:
  \[
    \epsilon_{kk'} \geq \epsilon_{\mathrm{lump}}
    \qquad\text{or}\qquad
    \phi_{kk'} = \frac{\Phi_{kk'}}{\displaystyle\sum_i \E[x_{ik}]}
    \;\geq\; \epsilon_{\mathrm{bn}}\cdot\bar\phi
  \]
  where $\bar\phi$ = mean per-molecule flux across the domain;
  $\epsilon_{\mathrm{lump}},\epsilon_{\mathrm{bn}}$ are user thresholds.

\end{frame}

\begin{frame}{Benchmark 1: Birth-Death-Diffusion (V-Cycle, $K=4$)}

  \textbf{Model:} $\emptyset \xrightarrow{k_b} A \xrightarrow{k_d} \emptyset$,
  $k_b=2$, $k_d=1$, $D=1$.
  \textbf{Smooth IC:} product-Poisson with $\mu=[4,2,1,1]$ per voxel.

  \medskip
  $K=4$ fine, $K=2$ coarse. Full V-cycle with Binomial-$\pi$ prolongation.
  Ground truth: $\pi^*$ (product-Poisson stationary distribution).

  \bigskip
  \begin{center}
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{l|c|c}
    \toprule
    Metric & Fine FSP & MG V-cycle \\
    \midrule
    $\TV(p,\,\pi^*)$ at $t=0.5,1,2,\ldots$ &
      $\searrow 0$ & $\mathbf{0.000}$ at all $t$ \\
    State space $|\mathcal{S}|$ & ${\sim}10{,}400$ & ${\sim}190$ \\
    Wall time & $5.3\,\text{s}$ & $\mathbf{0.22\,\text{s}}$ \\
    \bottomrule
  \end{tabular}
  \end{center}

  \bigskip
  \begin{block}{Exact coarse generator for linear networks}
    For birth-death-diffusion the Petrov--Galerkin term
    $\calR A_{\mathrm{rxn}} \calP$ is \emph{exact}.
    The V-cycle therefore matches $\pi^*$ \textbf{perfectly} at every time step.\\[0.35em]
    \textbf{54$\times$ fewer states.  \quad  24$\times$ faster.}
  \end{block}

\end{frame}

\begin{frame}{Three Prolongation Approaches}

  Coarse variable $n_c = n_1+n_2$ loses directionality.
  Three ways to reconstruct $p(n_1\mid n_c)$:

  \medskip
  \begin{description}
    \setlength\itemsep{0.75em}
    \item[\textbf{Binomial-$\pi$}:]
      $\pi(n_1\mid n_c) = \Bin(n_c,\tfrac12)$.
      Free to compute; ignores all reaction structure.
      \textcolor{slowred}{Symmetric — peaks at $n_1=n_c/2$.}

    \item[\textbf{Dynamic-$\pi$}:]
      True stationary conditional of the $K=2$ RDME,
      $\pi^*(n_1\mid n_c)$, precomputed once.
      Bistable: \emph{bimodal} at $(n_{\mathrm{low}},n_{\mathrm{high}})$
      and $(n_{\mathrm{high}},n_{\mathrm{low}})$ --- still 50/50 weight.
      \textcolor{slowred}{Symmetric — can't distinguish which voxel holds which state.}

    \item[\textbf{Fix-3 (fine-conditional injection):}]
      For states already represented in the fine distribution,
      rescale multiplicatively (inject, don't split).
      For newly created coarse states, fall back to $\pi^*(\cdot\mid n_c)$.
      \textcolor{coarsegreen}{Preserves the directionality of the front.}
  \end{description}

\end{frame}

\begin{frame}{Prolongation Failure: The Symmetry Trap}

  \textbf{Setup:} $K=2$, IC $(n_1,n_2)=(31,162)$.
  Coarse variable $n_c = 193$.

  \bigskip
  \begin{center}
    \includegraphics[height=4.6cm]{fig2_prolongation_failure}
  \end{center}

  \smallskip
  \begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Binomial-$\pi$:}
      $\Bin(193,\tfrac12)$ peaks at $n_1=96$ (midpoint). TV$=1.000$.
    \item \textbf{Dynamic-$\pi$:}

  \end{itemize}

\end{frame}

  Use it directly instead of re-splitting via Binomial.

\begin{frame}{Summary}

  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Method}
      \begin{itemize}
        \setlength\itemsep{0.45em}
        \item Spatial coarsening via lumpability
          \citeref{[Kemeny \& Snell 1960]};
          $\epsilon = d_{\mathrm{inter}}/d_{\mathrm{intra}} = 1/4$ per level
        \item V-cycle \citeref{[Briggs et al.\ 2000]}:
          smooth $\to$ restrict $\to$ coarse FSP $\to$ prolong $\to$ smooth
        \item Three prolongation operators:
          Binom-$\pi$, Dyn-$\pi$, Fix-3
        \item \textbf{Fix-3}: multiplicative injection preserves
          directionality of spatial fronts
        \item Two-criterion adaptive refinement:
          lumpability + per-molecule bottleneck
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Results}
      \begin{itemize}
        \setlength\itemsep{0.4em}
        \item Birth-death: TV$(p_{\mathrm{MG}},\pi^*)=0$ at all times
          (exact coarse generator); \textbf{24$\times$ faster}, 54$\times$ fewer states
        \item Schl\"ogl $K=4$: Fix-3 \textbf{360$\times$ faster} than Binom-$\pi$,
          \textbf{6$\times$ faster} than direct FSP; 221 vs 2M states
        \item Schl\"ogl $K=8$: Fix-3 solves in \textbf{1.7\,s},
          8361 states; direct FSP and Binom-$\pi$ both intractable
        \item Binom-$\pi$ / Dyn-$\pi$: symmetry trap, TV$\approx1$
          for asymmetric fronts
      \end{itemize}

      \medskip
      \textbf{Open}
      \begin{itemize}
        \setlength\itemsep{0.35em}
        \item Multi-level Fix-3 ($K\to K/2\to K/4\to\cdots$)
        \item Adaptive spatial refinement driven by per-molecule flux
      \end{itemize}
    \end{column}
  \end{columns}

\end{frame}
