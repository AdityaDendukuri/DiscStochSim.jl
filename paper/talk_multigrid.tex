\begin{frame}{The Reaction-Diffusion Master Equation}

  \textbf{Setting:}
  Domain $\Omega$ partitioned into $K$ voxels of size $h$.
  State $\bx_k$ = molecule counts in voxel $k$.

  \medskip
  \begin{center}
  \begin{tikzpicture}[scale=1.05]
    \foreach \k in {1,2,3,4} {
      \draw[thick,fill=blue!10] (\k-1,0) rectangle (\k,0.85);
      \node[font=\normalsize] at (\k-0.5,0.42) {$\bx_\k$};
    }
    \foreach \k in {1,2,3} {
      \draw[<->,fastblue,very thick] (\k-0.3,1.2) -- (\k+0.3,1.2);
    }
    \node[fastblue,font=\small] at (2,1.55)
      {diffusion: rate $d = D/h^2$ per molecule};
    \draw[->,slowred,thick] (2.5,0) -- (2.5,-0.42);
    \node[slowred,font=\small] at (2.5,-0.62)
      {reactions: propensity $\alpha_r(\bx_k)$ in each voxel};
  \end{tikzpicture}
  \end{center}

  \[
    \frac{dp}{dt} = A_{\mathrm{RDME}}\, p
    \;=\; \Bigl(\underbrace{A_{\mathrm{rxn}}}_{\text{slow}}
          + \underbrace{A_{\mathrm{diff}}}_{\text{fast}}\Bigr) p
  \]

  \begin{block}{FSP bottleneck \citeref{[Munsky \& Khammash 2006]}}
    State space $|\mathcal{S}| \sim n_{\max}^K$ grows exponentially in $K$.\\[0.2em]
    \begin{tabular}{cccc}
      $K=4$: & $13^4 \approx 28\text{k}$ & \textcolor{coarsegreen}{\small tractable} &\\
      $K=8$: & $13^8 \approx 8 \times 10^8$ &
        \textcolor{slowred}{\small intractable} &\hspace{0.5em}
    \end{tabular}
  \end{block}

\end{frame}

\section{Multigrid}
% ================================================================

\begin{frame}{Spatial Coarsening via Lumpability}

  \textbf{Idea:} merge adjacent fine-voxel pairs into a single coarse voxel.
  Coarse count: $\bar x_j = x_{2j-1} + x_{2j}$.

  \medskip
  \begin{center}
  \begin{tikzpicture}[scale=0.95]
    \foreach \k/\col in {1/blue!15,2/blue!15,3/red!15,4/red!15} {
      \draw[thick,fill=\col] (\k-1,1.2) rectangle (\k,2.0);
    }
    \node[font=\small] at (0.5,1.6) {$n_{\mathrm{lo}}$};
    \node[font=\small] at (1.5,1.6) {$n_{\mathrm{hi}}$};
    \node[font=\small] at (2.5,1.6) {$n_{\mathrm{hi}}$};
    \node[font=\small] at (3.5,1.6) {$n_{\mathrm{hi}}$};
    \node[left,font=\small] at (0,1.6) {fine:};
    \draw[thick,fill=blue!35] (0,0) rectangle (2,0.8);
    \draw[thick,fill=red!35]  (2,0) rectangle (4,0.8);
    \node[font=\small] at (1.0,0.4) {$\bar x_1 = n_{\mathrm{lo}}+n_{\mathrm{hi}}$};
    \node[font=\small] at (3.0,0.4) {$\bar x_2 = 2n_{\mathrm{hi}}$};
    \node[left,font=\small] at (0,0.4) {coarse:};
    \draw[->,thick,gray] (1.0,1.2) -- (1.0,0.8)
      node[midway,right,font=\footnotesize]{$\calR$: sum};
    \draw[->,thick,coarsegreen!80!black] (0.7,0.8) -- (0.7,1.2)
      node[midway,left,font=\footnotesize]{$\calP$: split};
  \end{tikzpicture}
  \end{center}

  \medskip
  \textbf{Why it works:}
  intra-pair diffusion at rate $D/h^2$ equilibrates molecules within a pair
  \emph{much faster} than inter-pair dynamics.
  After this fast equilibration the within-pair split follows:
  \[
    \pi(m \mid \bar x) = \Bin\!\bigl(\bar x,\,\tfrac{1}{2}\bigr)
  \]
  \textbf{Coarse generator} $\bar A = \calR A_{\mathrm{RDME}} \calP$
  is itself an RDME on a $K/2$-voxel grid.

\end{frame}

% ----------------------------------------------------------------
\begin{frame}{Two-Level V-Cycle}

  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{$p(t) \;\to\; p(t+\Delta t)$}
        \begin{enumerate}\setlength\itemsep{0.4em}
          \item \textbf{Restrict:}
            $p^c \gets \calR\, p$ \\
            {\small marginalize over within-pair splits}
          \item \textbf{Coarse solve:}
            $\bar p^c \gets e^{\Delta t\,\bar A}\, p^c$ \\
            {\small full time step on the small $K/2$ problem}
          \item \textbf{Prolong:}
            $p_{\mathrm{new}} \gets \calP\,\bar p^c$ \\
            {\small reconstruct fine distribution via $\Bin(\bar x, \tfrac{1}{2})$}
        \end{enumerate}
      \end{block}

      \smallskip
      \textbf{Why it's fast:}\\
      Coarse state space is $\sim n_{\max}^{K/2}$ vs.\ $n_{\max}^K$.\\[0.3em]
      $K{=}8{\to}4$: $\mathbf{13^4}$ vs.\ $\mathbf{13^8}$
      \textbf{(30k$\times$ smaller)}.
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{center}
      \begin{tikzpicture}[every node/.style={font=\footnotesize}, scale=0.82]
        \node[draw,rounded corners,fill=blue!22,
              minimum width=2.8cm,minimum height=0.62cm]
          (f) at (0,3.2) {fine ($K$ voxels)};
        \node[draw,rounded corners,fill=coarsegreen!28,
              minimum width=2.8cm,minimum height=0.62cm]
          (c) at (0,1.5) {coarse ($K/2$ voxels)};
        \draw[->,thick] (f.south west) -- (c.north west)
          node[midway,left]{$\calR$};
        \draw[->,thick,slowred] (c.north east) -- (f.south east)
          node[midway,right]{$\calP$};
        \node[coarsegreen!70!black] at (0,0.9){FSP solve};
        \draw[coarsegreen!60!black,thick,rounded corners]
          (-1.2,0.62) rectangle (1.2,1.15);
      \end{tikzpicture}
      \end{center}

      \smallskip
      {\small\textbf{Analogy to geometric MG:}\\
      Restrict $=$ marginalize\\
      Coarse solve $=$ lumped CME\\
      Prolong $=$ Binomial split\\
      Smooth $=$ intra-pair diffusion}
    \end{column}
  \end{columns}

\end{frame}

\begin{frame}{The Symmetry Trap}

  \textbf{Problem:} a spatial front lives \emph{inside} a coarse voxel.

  \medskip
  \begin{center}
  \begin{tikzpicture}[scale=0.9, every node/.style={font=\small}]
    % Fine level
    \draw[thick,fill=blue!20] (0,2.2) rectangle (1.6,3.0);
    \node at (0.8,2.6) {$n_1 = 31$};
    \draw[thick,fill=red!20] (1.6,2.2) rectangle (3.2,3.0);
    \node at (2.4,2.6) {$n_2 = 162$};
    \draw[thick,fill=red!15] (3.2,2.2) rectangle (4.8,3.0);
    \node at (4.0,2.6) {$n_3 = 162$};
    \node[left,font=\footnotesize] at (0,2.6) {fine:};

    % Restrict arrow
    \draw[->,thick,gray] (1.6,2.2) -- (1.0,1.5)
      node[midway,left,font=\footnotesize]{$\calR$};

    % Coarse level
    \draw[thick,fill=blue!35] (0,0.6) rectangle (3.2,1.4);
    \node at (1.6,1.0) {$\bar n_1 = 31+162 = 193$};
    \draw[thick,fill=red!25] (3.2,0.6) rectangle (4.8,1.4);

    \node[left,font=\footnotesize] at (0,1.0) {coarse:};

    % Wrong result
    \draw[thick,fill=yellow!30] (0,-0.9) rectangle (1.6,-0.1);
    \node at (0.8,-0.5) {$n_1 \approx 96$\;\textcolor{slowred}{\ding{55}}};
    \draw[thick,fill=yellow!30] (1.6,-0.9) rectangle (3.2,-0.1);
    \node at (2.4,-0.5) {$n_2 \approx 97$\;\textcolor{slowred}{\ding{55}}};
    \node[left,font=\footnotesize] at (0,-0.5) {result:};

    % Annotation
    \node[slowred,align=center,font=\footnotesize] at (6.2,1.0)
      {$\Bin(193,\tfrac12)$ peaks\\at midpoint 96\\[0.2em]\textbf{front is erased}};
  \end{tikzpicture}
  \end{center}

  \begin{alertblock}{Root cause}
    The coarse variable $\bar n_1 = 193$ is consistent with both
    $(31, 162)$ \emph{and} $(162, 31)$.
    Binomial splits them 50/50 around the midpoint.
    \textbf{Directionality is lost.}
  \end{alertblock}

\end{frame}

\begin{frame}{The Fix: Don't Coarsen the Front}

  \begin{block}{Key insight}
    The Binomial approximation is only wrong \emph{where the distribution
    is asymmetric within a pair}.
    For bulk pairs far from the front it works well.
    \textbf{Solution: selectively refine only the pairs that need it.}
  \end{block}

\begin{frame}[shrink=3]{Detecting Which Pairs Need Refinement}

  Two admissibility diagnostics, computed from voxel means $\mu_k = \E[n_k]$:

  \smallskip
  \begin{description}\setlength\itemsep{0.6em}
    \item[\textbf{Within-pair asymmetry} $\alpha_j$:]
      \[
        \alpha_j = \frac{|\mu_{2j-1} - \mu_{2j}|}{\mu_{2j-1}+\mu_{2j}}
      \]
      Large when \emph{the front lies inside pair $j$}
      (one voxel high, other low).

    \item[\textbf{Inter-pair boundary gradient} $\beta_j$:]
      \[
        \beta_j = \frac{|\mu_{2j} - \mu_{2j+1}|}{\mu_{2j}+\mu_{2j+1}}
        \qquad (j < K/2)
      \]
      Large when \emph{the front lies on the boundary between pairs $j$ and $j+1$}.
      Forces both pairs $j$ and $j+1$ fine.
  \end{description}

  \smallskip
  \begin{block}{Refinement rule (with hysteresis)}
    Define $\alpha^\mathrm{eff}_j = \max(\alpha_j,\;\beta_{j-1},\;\beta_j)$
    \;{\footnotesize(boundary terms $\beta_0 = \beta_{K/2} = 0$)}.
    \[
      \alpha^\mathrm{eff}_j > \alpha_{\mathrm{hi}} \;\Rightarrow\; \text{force fine};
      \qquad
      \alpha^\mathrm{eff}_j < \alpha_{\mathrm{lo}} \;\Rightarrow\; \text{coarsen}.
    \]
    Between the bands: keep previous mask (prevents chatter at moving fronts).
  \end{block}

\end{frame}

\begin{frame}{The Mixed-Grid CME}

  Once the mask is chosen, we need a generator for the mixed-dimension
  state space $\mathcal{S}^M \subset \Znn^{K_M}$.

  \medskip
  \textbf{Mixed-grid rates:}
  \begin{itemize}\setlength\itemsep{0.5em}
    \item \textbf{Fine pair} (mask$_j=0$): standard per-voxel Schlögl rates;
      intra-pair diffusion included.
    \item \textbf{Coarse pair} (mask$_j=1$): reaction rates with Binomial
      first-moment closure (e.g.\ autocatalysis $\propto \bar n(\bar n-1)/4$
      for a merged pair of equal sub-voxels).
    \item \textbf{Cross-pair diffusion:}
      fine side $\to$ coarse side at rate $d\,n_{2j}$;\;
      coarse side $\to$ fine side at $d\,\bar n_j/2$
      (first-moment: $\E[n_{2j+1}\mid\bar n_j] = \bar n_j/2$).
  \end{itemize}

  \medskip
  \begin{block}{Why no multiplicative correction is needed}
    Front pairs are kept fine: their within-pair distribution is never
    approximated by Binomial. The coarse pairs are bulk pairs where
    the Binomial assumption holds by symmetry.
    \textbf{Binomial prolongation is then accurate where applied.}
  \end{block}

\end{frame}

\section{Summary}
% ================================================================

\begin{frame}[shrink=3]{Summary}

  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{What we built}
      \begin{itemize}\setlength\itemsep{0.45em}
        \item \textbf{Probabilistic two-level multigrid} for the CME.
          Restriction = exact marginalisation.
          Prolongation = Binomial conditional.
          Coarse operator = Galerkin-projected RDME.

        \item \textbf{Adaptive pair selection.}
          Two diagnostics: within-pair $\alpha_j$ and
          inter-pair $\beta_j$. Hysteresis prevents chatter.

        \item \textbf{Mixed-grid CME.}
          Fine and coarse pairs in a single generator;
          first-moment closure at cross-pair boundaries.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Key results}
      \begin{itemize}\setlength\itemsep{0.45em}
        \item Adaptive mask is \textbf{partition-independent}:
          detects fronts inside a pair ($\alpha$) \emph{and}
          on pair boundaries ($\beta$).

        \item Front pairs kept fine $\Rightarrow$
          \textbf{no symmetry trap}, no special correction needed.

        \item $K=8$ Schlögl: direct FSP intractable;
          adaptive V-cycle solves with modest state growth.
      \end{itemize}

      \medskip
      \textbf{Open directions}
      \begin{itemize}\setlength\itemsep{0.3em}
        \item Multi-level hierarchy ($K\to K/2\to\cdots$)
        \item Multi-species / 2D spatial domains
        \item Error-driven automatic threshold selection
      \end{itemize}
    \end{column}
  \end{columns}

\end{frame}
