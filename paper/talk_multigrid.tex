\begin{frame}{The problem: rare events in spatial stochastic kinetics}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      Reaction--diffusion on a lattice: molecules react within a voxel and hop to
      neighbours. At low copy number the noise is intrinsic.
      \begin{itemize}\setlength\itemsep{0.35em}
        \item Bistable chemistry gives \textbf{metastable} basins; the relevant
              transitions (basin switching, nucleation) are \textbf{rare}.
        \item Target: the stationary distribution \emph{and} the switching
              \textbf{rate}, not a single trajectory.
      \end{itemize}

      \medskip
      \begin{block}{This talk}\small
        One operator $A_c=RAP$ on \textbf{two axes}: count (basins) and space
        (voxel blocks). Solve the joint at scale, \emph{and} reduce it to a
        $\sim$200-state rate-faithful model.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \footnotesize
      \textbf{Limitations of existing tools:}
      \smallskip

      \begin{tabular}{@{}p{0.30\linewidth}p{0.62\linewidth}@{}}
        \toprule
        \textbf{approach} & \textbf{limitation} \\
        \midrule
        SSA (sampling) & rare switch requires prohibitively many samples;
                         $\lambda_2$ underflows \\[0.35em]
        Dense / direct FSP & joint is $n_{\max}^{K^2}$
                         ($200{\times}200\Rightarrow n_{\max}^{40000}$);
                         LU infeasible by 6.8M states \\[0.35em]
        Standard multigrid (uniform $P$) & coarsens, but corrupts the slow mode:
                         $150\%$ rate error \\
        \bottomrule
      \end{tabular}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The Chemical Master Equation}

  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      Distribution $p(n,t)$ over integer counts, one linear ODE:
      \[ \dot p = A\,p . \]
      \begin{itemize}\setlength\itemsep{0.3em}
        \item $\mathbf 1^\top A=0$ (mass-conserving); steady state $=$ null vector.
        \item $p$ is the full distribution: tails, multiple peaks, rare events.
      \end{itemize}

      \smallskip
      \begin{block}{Finite State Projection}\small
        Restrict to a finite support $\Sset$ and integrate
        \[ p(t{+}\Delta t)=e^{\Delta t\,A_\Sset}\,p\quad(\mathtt{expv}). \]
        Expand $\Sset$ where probability flows out, discard the tail. Cost $\sim|\Sset|$.
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Bistable birth--death (Schl\"ogl):}\\[1pt]
      {\small $2X \rightleftharpoons 3X,\quad \emptyset \rightleftharpoons X$ \ \
      (state-dependent $b(n),d(n)$)}

\begin{frame}{Restriction $R$: sum fine mass into cells}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Definition 1 (Partition \& restriction)}\small
        Partition the fine support into disjoint cells $\Sset=\bigsqcup_c \Sset_c$;
        each microstate $f\in\Sset$ lies in exactly one cell $c$. The restriction
        $R\in\mathbb R^{|\mathcal C|\times|\Sset|}$ is
        \[
          R_{cf}=\mathbf 1[f\in c],\qquad
          (R\,p)_c=\sum_{f\in c}p_f .
        \]
      \end{block}

      {\small The coarse variable is the cell label. A cell is \emph{any} group of
      microstates: a count-basin, a block of voxels, an aggregate. The example
      below uses three arbitrary cells $c_1,c_2,c_3$.}

% ================================================================
%  A2b.  Prolongation P and the Galerkin operator A_c = R A P
% ================================================================
\begin{frame}{Prolongation $P$: distribute by the within-cell conditional}

  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Definition 2 (Prolongation)}\small
        Fix within-cell weights $\varphi_f=\pi(f\mid c)$, the \emph{stationary
        conditional} ($\sum_{f\in c}\varphi_f=1$). Then
        $P\in\mathbb R^{|\Sset|\times|\mathcal C|}$ is
        \[
          P_{fc}=\varphi_f\,\mathbf 1[f\in c],\qquad
          (P\,q)_f=q_c\,\varphi_f\ \ (f\in c).
        \]
        \emph{Galerkin coarse generator} $A_c\coloneqq R\,A\,P$.
      \end{block}

      \begin{block}{Lemma 1 (Consistency)}\small
        $R\,P=I_{|\mathcal C|}$ and $\mathbf 1^\top A_c=0$. ($\|RP-I\|=2.2\mathrm{e}{-16}$.)
      \end{block}

      {\footnotesize Taking $\varphi=\pi(\cdot\mid c)$ is the
      \textbf{structure-preserving} (stationary-weighted) choice: it restores each
      cell's conditional shape, so multimodal structure survives $P\!\circ\!R$. Uniform
      weights $\varphi_f=1/|\Sset_c|$ do \emph{not} preserve the slow dynamics
      (later: the rate).}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[>=stealth,scale=0.74,every node/.style={font=\tiny}]

\begin{frame}{Operator splitting and the V-cycle}

  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \[ A=A_{\rm fast}+A_{\rm slow} \]
      \begin{itemize}\setlength\itemsep{0.3em}
        \item $A_{\rm fast}$: within-basin relaxation, large $|\lambda|$
              $\to$ \textbf{smoother} $S$
        \item $A_{\rm slow}$: basin switching $\to$ \textbf{coarse grid} $A_c$
      \end{itemize}
      \smallskip
      $S$ damps the fast error from $A_{\rm fast}$; the coarse solve on $A_c$
      corrects the slow error left behind. The V-cycle alternates the two,
      for the backward-Euler step $(I-\Delta t\,A)x=b$:
      \begin{enumerate}\setlength\itemsep{0.18em}
        \item pre-smooth $x\leftarrow S^{\nu}x$
        \item residual $r=b-(I-\Delta t\,A)x$
        \item coarse solve $(I-\Delta t\,A_c)\,e_c=R\,r$
        \item correct $x\leftarrow x+P\,e_c$
        \item post-smooth $x\leftarrow S^{\nu}x$
      \end{enumerate}
      Recurse on $A_c$ for a full V-cycle.
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{center}
      \begin{tikzpicture}[>=stealth,scale=0.86,every node/.style={font=\scriptsize}]
        \foreach \y/\lbl/\col in {2/L0/myblue,1/L1/{intercolor!80!black},0/L2/{coarsegreen!65!black}}{

\begin{frame}{The spatial problem: the reaction--diffusion master equation}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      $K\times K$ grid; voxel $\mathbf k$ holds count $n_\mathbf{k}$. Diffusion is a
      \textbf{first-order (linear) reaction} $X_\mathbf{k}\to X_\mathbf{l}$ at rate
      $d=D/h^2$, so space coarsening is \emph{exact} (Theorem 1).
      \[
        A=\underbrace{\textstyle\sum_\mathbf{k} A^{\rm rx}_\mathbf{k}}_{\text{reaction (diag.)}}
         +\underbrace{\textstyle\sum_{\langle\mathbf k,\mathbf l\rangle} A^{\rm diff}_{\mathbf k\mathbf l}}_{X_\mathbf{k}\to X_\mathbf{l}\ \text{(off-diag.)}}
      \]
      \begin{itemize}\setlength\itemsep{0.35em}
        \item \textbf{Joint} law, up to $n_{\max}^{K^2}$; a $200{\times}200$ grid is
              $n_{\max}^{40000}$, never assembled.
        \item Neighbour-only coupling $\Rightarrow$ smooth, near-steady away from the front.
      \end{itemize}

      \begin{alertblock}{Role of the coarse operator}
        One operator $A_c=RAP$ serves two purposes: it solves the joint at scale,
        and reduces the model to $\sim$200 states that preserve the switching rate.
      \end{alertblock}
    \end{column}

% ================================================================
\begin{frame}{The metastable rate is an eigenvalue}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      Spectrum of $A$: $0=\lambda_1>\mathrm{Re}\,\lambda_2\ge\cdots$; the stationary
      law $\pi$ is the $\lambda_1{=}0$ mode.
      \begin{itemize}\setlength\itemsep{0.35em}
        \item Bistability gives a \textbf{spectral gap}: one eigenvalue $\lambda_2$
              lies near $0$, isolated from the bulk.
        \item Its mode is the basin imbalance; relaxation
              $p(t)-\pi\sim e^{\lambda_2 t}$.
        \item $|\lambda_2|$ \emph{is} the basin-switching rate,
              $\approx 1/\langle T_{\rm switch}\rangle$: the rare event SSA cannot
              sample.
      \end{itemize}

      \end{block}

\begin{frame}{Exact for linear reactions; vanishing error otherwise}

  \begin{block}{Theorem 1 (Exact coarsening for linear networks)}\small
    If every reaction propensity is affine in the counts and diffusion is linear,
    then the within-cell conditional $\varphi=\pi(\cdot\mid c)$ is exactly
    (multi)nomial and the Galerkin operator $A_c=R\,A\,P$ generates the coarse
    variable \emph{exactly}: the closure error is zero.
  \end{block}

  \begin{columns}[T]
    \begin{column}{0.62\textwidth}
      \begin{center}
        \includegraphics[width=\linewidth]{fig_closure_highcopy}
      \end{center}
    \end{column}
    \begin{column}{0.36\textwidth}
      {\small
      \textbf{Nonlinear $\alpha$.} For a smooth propensity the leading closure error is
      \[
        \tfrac12\,\alpha''(\langle n\rangle)\,\mathrm{Var}(n)
        =O\!\big(1/\langle n\rangle\big)\xrightarrow[\text{high copy}]{}0 .
      \]
      \begin{itemize}\setlength\itemsep{0.4em}
        \item \textbf{Linear:} exact to machine precision, every copy number.
        \item \textbf{Nonlinear:} error $\propto 1/\langle n\rangle$.
        \item Coarsen \textbf{where counts are high} (bulk); keep the front fine.
      \end{itemize}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Adaptive refinement: two flux criteria}

  {\small\textbf{Space axis}: coarsen the bulk, keep the fine grid where the front moves.}
  \[
    \Phi_\mathbf{k}=D\,\langle n\rangle\ \ (\text{total flux}),\qquad
    \phi_\mathbf{k}=D\!\!\sum_{n>n_{\rm un}}\!\! n\,p(n)\ \ (\text{per-molecule flux})
  \]

  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      {\small
      \begin{itemize}\setlength\itemsep{0.5em}
        \item $\Phi$ misses a \textbf{bottleneck}: low mean, transition-competent
              tail.
        \item $\phi$ counts only molecules above $n_{\rm un}$, those that
              nucleate a neighbour.
        \item \textbf{refine if $\Phi$ \emph{or} $\phi>\mathrm{tol}$.}
      \end{itemize}}

% ================================================================
%  Summary
% ================================================================
\begin{frame}{Summary}
  \begin{itemize}\setlength\itemsep{0.7em}
    \item \textbf{One principle, two axes.} Merging voxels and grouping
          count-basins are the \emph{same} coarsening; multigrid applies either,
          recursively.
    \item \textbf{Known prolongation.} $P=$ the within-cell steady state;
          $A_c=RAP$ is \textbf{exact for linear networks}, with
          $O(1/\langle n\rangle)$ error at high copy number.
    \item \textbf{Adaptive grid.} Total \emph{and} per-molecule flux keep the front
          fine and the bulk coarse; $|\Sset|$ tracks the active band, not the domain.
    \item \textbf{Measured \& scalable.} A complete V-cycle: $1424\times$
          coarsening, $1.5\times10^{-4}$ vs direct; exact on the linear chain.
          As a GMRES preconditioner it is \textbf{size-independent}: $23\!\to\!32$
          iterations as $|\Sset|$ grows $2.6\mathrm{k}\!\to\!6.8\mathrm{M}$.
    \item \textbf{Scope.} Exact under timescale separation; degrades when
          correlations extend (a broad 2D interface), where the slow subspace is
          no longer low-dimensional.
  \end{itemize}
\end{frame}
