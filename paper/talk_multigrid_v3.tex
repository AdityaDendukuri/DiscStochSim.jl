\begin{frame}{The RDME: reactions and diffusion across a spatial grid}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Model.}
      Divide a 1-D domain into $K$ voxels.
      Each voxel $k$ holds molecule counts $\mathbf{n}_k \in \mathbb{Z}_{\geq 0}^S$.

      \medskip
      Two types of events:
      \begin{itemize}\setlength\itemsep{0.4em}
        \item \textcolor{slowred}{\textbf{Reactions}} inside voxel $k$:
              rate $\alpha_r(\mathbf{n}_k)$, change $\mathbf{n}_k$.
        \item \textcolor{fastblue}{\textbf{Diffusion}} between neighbors:
              rate $d = D/h^2$, moves one molecule $k \leftrightarrow k{+}1$.
      \end{itemize}

      \medskip
      The joint probability $p(\mathbf{n}_1,\ldots,\mathbf{n}_K,t)$
      satisfies $\dot p = A p$ \textbf{(RDME)}.

      \medskip
      \begin{alertblock}{The computational bottleneck}
        The state space has $\sim n_{\max}^{SK}$ reachable states ---
        \textbf{exponential in $K$}.
        Even $K{=}8$, $S{=}1$ is intractable with a naive FSP.
      \end{alertblock}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.90, every node/.style={font=\small}]
        \foreach \k in {1,2,3,4,5,6} {
          \draw[thick,fill=myblue!12] (\k-1,0) rectangle (\k,1.0);
          \node at (\k-0.5, 0.50) {$\mathbf{n}_\k$};
        }
        \foreach \k in {1,2,3,4,5} {
          \draw[<->,fastblue,thick] (\k-0.15,1.18) -- (\k+0.15,1.18);
        }
        \node[fastblue, font=\scriptsize] at (3, 1.45) {diffusion $d = D/h^2$};
        \foreach \k in {1,2,3,4,5,6} {
          \draw[->,slowred,thick] (\k-0.5,-0.05) -- (\k-0.5,-0.38);
        }
        \node[slowred, font=\scriptsize] at (3, -0.58) {reactions $\alpha_r(\mathbf{n}_k)$};
        \draw[<->,thick,mygray] (0,-1.1) -- (6,-1.1)
          node[midway,below,font=\scriptsize]{$K$ voxels};
        \node[mygray,font=\scriptsize] at (3,-1.65)
          {state space: $n_{\max}^{SK}$ entries};
      \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}

\end{frame}

\begin{frame}{The Binomial is the exact diffusion equilibrium}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Fix total molecules} $\bar n = n_1 + n_2$ across two adjacent voxels.

      \smallskip
      Diffusion moves one molecule at a time, at rate $d$ per molecule:

      \begin{center}
      \begin{tikzpicture}[every node/.style={font=\small}, scale=0.88]
        \draw[thick,fill=coarsecolor!15] (0,0) rectangle (1.3,0.85);
        \draw[thick,fill=coarsecolor!15] (2.8,0) rectangle (4.1,0.85);
        \node at (0.65, 0.42) {$n_1$};
        \node at (3.45, 0.42) {$n_2$};
        \draw[->,fastblue,thick,bend left=22] (1.3,0.58)
          to node[above,font=\scriptsize]{rate $d\cdot n_1$} (2.8,0.58);
        \draw[->,slowred,thick,bend left=22]  (2.8,0.27)
          to node[below,font=\scriptsize]{rate $d\cdot n_2$} (1.3,0.27);
      \end{tikzpicture}
      \end{center}

      Each molecule hops independently $\Rightarrow$ at equilibrium, equally
      likely to be in either voxel. For $\bar n$ total:
      \[
        \boxed{P(n_1 \mid \bar n) = \Bin\!\bigl(\bar n,\,\tfrac{1}{2}\bigr)}
        \quad \text{(exact, any } d > 0\text{)}
      \]
      When the distribution is near this equilibrium,
      \textbf{$\bar n$ alone describes the two-voxel state}.
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.85]

\begin{frame}{Merging two adjacent voxels: restriction and prolongation}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Compress ($\calR$): exact:}
      \[ p^c(\bar n) = \!\!\sum_{n_1 + n_2 = \bar n}\!\! p(n_1, n_2). \]

      \textbf{Reconstruct ($\calP$): approximate:}
      \[ p(n_1, n_2) \approx p^c(\bar n)\cdot\Bin(\bar n,\tfrac{1}{2}). \]

      \begin{block}{State-space reduction}
        Per pair: $n_{\max}^{2S} \to n_{\max}^{S}$.\\
        All $K$ voxels: $n_{\max}^{SK} \to n_{\max}^{SK/2}$.
      \end{block}

      \smallskip
      {\small $\calP$ is accurate when the distribution is near-Binomial}
    \end{column}

\begin{frame}{Bistable reactions}

\begin{columns}[T]
\column{0.50\textwidth}
For birth-death, each voxel has one stable state.
Diffusion wins: the pair is always well-mixed.
Binomial coarsening is \emph{essentially exact}.

\vspace{0.8em}
\textbf{But what if reactions are bistable?}\\[0.4em]
The \textbf{Schl\"{o}gl model} has two stable states per voxel:
$n_\text{lo} \approx 31$ and $n_\text{hi} \approx 162$.

\vspace{0.4em}
\begin{center}
  \includegraphics[width=0.95\textwidth]{fig_talk_bistability}
\end{center}

\column{0.46\textwidth}
\vspace{0.3em}
\begin{block}{Spatial metastability}
Neighbors can occupy \emph{different} states.
A \textbf{spatial front} forms.
\end{block}

\medskip
\begin{tikzpicture}[scale=0.7, every node/.style={font=\footnotesize}]
  \foreach \k in {1,2} {
    \draw[thick,fill=myblue!22] (\k-1,0) rectangle (\k,0.8);
    \node at (\k-0.5,0.4) {lo};
  }
  \foreach \k in {3,4} {
    \draw[thick,fill=slowred!22] (\k-1,0) rectangle (\k,0.8);
    \node at (\k-0.5,0.4) {hi};
  }
  \draw[very thick,intercolor,dashed] (2,-0.12) -- (2,0.96);
\end{tikzpicture}

\medskip
{\small Reactions ``pull'' voxels apart; diffusion ``pulls'' them together.
When reactions are fast, the front \textbf{persists for very long times}.}
\end{columns}

\end{frame}

\begin{frame}{Why Binomial prolongation fails at bistable fronts}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      At a spatial interface, the joint distribution $P(n_1, n_2)$
      is \textbf{bimodal}: mass concentrated at $(n_\text{lo}, n_\text{hi})$
      and $(n_\text{hi}, n_\text{lo})$.

      \medskip
      \begin{alertblock}{Restriction loses orientation}
        $\calR(n_\text{lo}, n_\text{hi}) = n_\text{lo}+n_\text{hi} = \calR(n_\text{hi}, n_\text{lo})$.\\[0.3em]
        The total $\bar n = 193$ is the same whether the low-state voxel
        is on the left or the right.
        \textbf{The front direction is erased.}
      \end{alertblock}

      \medskip
      Binomial prolongation then places the reconstructed mass at
      $(n_\text{hi}/2{+}n_\text{lo}/2,\;\ldots) \approx (96, 97)$ ---
      squarely in the \textbf{unstable} region.
      The metastable front collapses to an unphysical average.
    \end{column}

    \begin{column}{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.9]
        \draw[->] (0,0) -- (4.2,0) node[right,font=\small]{$n_1$};
        \draw[->] (0,0) -- (0,4.2) node[above,font=\small]{$n_2$};
        % bimodal blobs

        % Binomial reconstruction arrow to unstable center

      }

\end{frame}


\begin{frame}{The $n$-level recursive V-cycle}

\begin{columns}[T]
\column{0.54\textwidth}

\textbf{One V-cycle step at level $K$:}\\[0.4em]

\begin{enumerate}\setlength\itemsep{0.28em}
  \item \textbf{Pre-smooth}: evolve intra-pair diffusion for $\tau_{\rm pre}$
        $\to$ drives within-pair distribution toward
        $\Bin(\bar n_j,\tfrac{1}{2})$

  \item \textbf{Restrict}: $\bar n_j = n_{2j-1}+n_{2j}$;
        save $\bar p^{\,\rm pre} = \calR\,\tilde p$ (pre-solve coarse distribution)

  \item \textbf{Recurse}: V-cycle on $K/2$-voxel system\\
        (direct FSP at the coarsest level $K/2^L$)

  \item \textbf{Prolong}: conditional reconstruction
        from coarse update $\bar p^{\,\rm post}$

  \item \textbf{Prune}: remove states with $p(x) < \tau$
\end{enumerate}

\column{0.42\textwidth}
\vspace{0.4em}
\begin{block}{Hierarchy}
  $K \;\to\; K/2 \;\to\; K/4 \;\to\; \cdots \;\to\; K/2^L$\\[0.3em]
  Coarsest level solved exactly with the FSP.
  Each level adds one restriction + prolongation step.
\end{block}

\vspace{0.4em}
\begin{block}{State space}
  Grows \emph{only} through prolongation --- no explicit enumeration
  of the full $n_{\max}^K$ space.
  Support of $p$ tracked directly; controlled by the prune threshold~$\tau$.
\end{block}

\end{columns}

\end{frame}


\begin{frame}{Conditional prolongation: reconstructing the fine distribution}

\begin{columns}[T]
\column{0.56\textwidth}

After the coarse solve, $\bar p^{\rm post}$ differs from $\bar p^{\rm pre}$.
Three types of coarse states arise:

\medskip
\textbf{1.~Covered states}
  $\bigl(\bar x \in \mathcal{S}^{\rm pre} = \calR\,\tilde p\bigr)$:
  \textbf{multiplicative update}
  \[
    p^{\rm new}(x) \;=\; \tilde p(x)\cdot
    \frac{\bar p^{\rm post}(\bar x)}{\bar p^{\rm pre}(\bar x)}.
  \]
  Exact and asymmetry-preserving: mass at $(n_{\rm lo},n_{\rm hi})$
  stays there rather than being forced symmetric.

\medskip
\textbf{2.~Reaction boundary} ($L_1 = 1$: one pair count changes by ${\pm}1$):\\
  Extend the nearest covered conditional --- keep $n_{2j-1}$,
  set $n_{2j} = \bar n_j^{\rm new} - n_{2j-1}$.

\medskip
\textbf{3.~Diffusion boundary} ($L_1 = 2$: two pair counts each change by ${\pm}1$):\\
  Coarse diffusion moves a molecule between pairs simultaneously.
  Shift second voxels of \emph{both} changed pairs.

\column{0.40\textwidth}
\vspace{0.5em}

\begin{center}
\begin{tikzpicture}[scale=0.78,every node/.style={font=\scriptsize}]
  % Coarse grid K=2: nc1, nc2
  \node[above] at (1.0,2.2) {$(\bar n_1,\bar n_2)$ coarse state};
  % covered state
  \fill[coarsecolor!30] (0,1) rectangle (2,2);
  \node at (1,1.5) {covered $\bar x$};
  % reaction boundary L1=1
  \fill[coarsegreen!25] (0,0) rectangle (2,1);
  \node at (1,0.5) {$L_1{=}1$: reaction $(\bar n_1{\pm}1)$};
  % diffusion boundary L1=2
  \fill[intercolor!25] (2,0) rectangle (4,1);
  \node at (3,0.5) {$L_1{=}2$: diffusion};
  \node[align=center] at (3,1.5) {$(\bar n_1{-}1,\,\bar n_2{+}1)$\\both pairs shift};
  \fill[intercolor!25] (2,1) rectangle (4,2);
  \draw[thick] (0,0) rectangle (4,2);
  \draw[thick] (0,1)--(4,1);
  \draw[thick] (2,0)--(2,2);
  % arrow labels
  \draw[->,thick,coarsecolor] (-0.8,1.5) -- (0,1.5) node[left,pos=0]{$\times$-update};
  \draw[->,thick,coarsegreen!70!black] (-0.8,0.5) -- (0,0.5) node[left,pos=0]{extend};

\end{tikzpicture}
\end{center}

\vspace{0.3em}
\begin{alertblock}{Why $L_1{=}2$ matters}
  Coarse diffusion $({\bar n}_1,{\bar n}_2)\!\to\!({\bar n}_1{-}1,{\bar n}_2{+}1)$
  is always $L_1{=}2$.
  Without this search, \textbf{inter-pair mass transport is invisible} to
  the V-cycle and molecules never leave pair~1.
\end{alertblock}

\end{columns}

\end{frame}

\begin{frame}{Summary}

  \begin{block}{1.\quad Fast diffusion enables state-space compression via restriction.}
    When the within-pair distribution is near-Binomial, marginalising pairs reduces
    the state space from $n_{\max}^{2S}$ to $n_{\max}^{S}$ per pair.
    \emph{$K{=}8$ birth-death wavefront solved with 30k states (direct: intractable).}
  \end{block}

  \smallskip
  \begin{block}{2.\quad Bistable systems require conditional --- not Binomial --- prolongation.}
    At spatial fronts, the within-pair distribution is bimodal.
    Conditional prolongation preserves asymmetry: multiplicative update for covered states;
    $L_1{\leq}2$ neighbor search handles both reaction ($L_1{=}1$) and
    diffusion ($L_1{=}2$) boundary states.
  \end{block}

  \smallskip
  \begin{block}{3.\quad The $n$-level V-cycle propagates correct spatial dynamics.}
    Wavefront spreading (from concentrated IC) and bistable splitting (from unstable IC)
    are both captured.
    State-space growth is controlled by pruning: no explosion.
  \end{block}

  \smallskip
  \textbf{Open questions:} error bounds on conditional prolongation;
  dynamic-$\pi$ for improved within-pair accuracy; multi-species Schl\"{o}gl systems.

\end{frame}
