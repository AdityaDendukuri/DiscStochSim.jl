\begin{frame}{The Chemical Master Equation: exact stochastic distributions}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      A well-mixed system: integer counts
      $\mathbf{n} \in \mathbb{Z}_{\geq 0}^S$ evolving by stochastic reactions.

      \medskip
      The probability $p(\mathbf{n}, t)$ satisfies the \textbf{CME}:
      \[
        \dot{p} = A\,p,
      \]
      where $A$ encodes every reaction event as a transition rate.

      \bigskip
      \begin{alertblock}{SSA vs.\ CME}
        \textbf{SSA} samples individual trajectories.\\[0.3em]
        \textbf{CME} gives the \emph{exact} joint distribution
        $p(\mathbf{n},t)$ directly --- rare events, tails, and
        multimodal structure without ensemble noise.
      \end{alertblock}

      \bigskip
      {\small
        \textbf{FSP}: truncate $A$ to a finite support $\mathcal{S}$ and solve
        $\dot p = Ap$ numerically.  Expand $\mathcal{S}$ at the boundary;
        prune negligible states.
      }
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{Birth-death: $\emptyset \xrightarrow{\lambda} X \xrightarrow{\mu n} \emptyset$}

      \smallskip
      \begin{center}

                          >=stealth]

\begin{frame}{The RDME: reactions \emph{and} diffusion on a spatial grid}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      Divide a 2-D domain into $K{\times}K$ voxels.
      Each voxel $\mathbf{k}$ holds a molecule count $n_{\mathbf{k}} \in \mathbb{Z}_{\geq 0}$.

      \medskip
      Two types of events:
      \begin{itemize}\setlength\itemsep{0.4em}
        \item \textcolor{slowred}{\textbf{Reactions}} inside voxel $\mathbf{k}$:
              propensity $\alpha(n_{\mathbf{k}})$.
        \item \textcolor{fastblue}{\textbf{Diffusion}} between neighbors $\mathbf{k}{\leftrightarrow}\mathbf{l}$:
              rate $d = D/h^2$, moves one molecule.
      \end{itemize}

      \medskip
      The joint probability $p(\{n_{\mathbf{k}}\},t)$ satisfies
      $\dot p = A p$ \textbf{(RDME)}.

      \medskip
      \begin{alertblock}{The curse of dimensionality}
        Joint state space: ${\sim}n_{\max}^{K^2}$ entries.\\
        A $200{\times}200$ grid is \textbf{completely intractable} directly.
      \end{alertblock}

      \medskip
      {\small
        \textbf{Key locality:} diffusion only couples \emph{adjacent} voxels.
        This structure enables a principled decomposition.
      }
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.78, every node/.style={font=\scriptsize}]

\begin{frame}{Per-voxel CME with mean-field diffusion coupling}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Key idea:} diffusion couples voxels through their means.

      \medskip
      A molecule hops from voxel $\ell$ to voxel $k$ at rate $d\cdot n_\ell$.
      Replace $n_\ell$ by its mean $\mu_\ell = \langle n_\ell \rangle$:

      \medskip
      \begin{block}{Mean-field RDME decomposition}
        \vspace{-0.3em}
        \begin{align*}
          \text{Diffusion in voxel }k\text{:}&\quad \text{birth at rate}\;\textstyle d\sum_{\ell\sim k}\mu_\ell\\
          &\quad \text{death at rate}\;d\cdot|\partial k|\cdot n_k
        \end{align*}
        Each voxel $k$ evolves under an \textbf{effective 1-D CME}:
        \[
          \dot p_k = Q_k(\{\mu_\ell\}_{\ell\sim k})\,p_k
        \]
      \end{block}

      \medskip
      {\small
        \textbf{Why it works:} at the wavefront, equil voxels are sharp
        (delta-like at their basin), so $\mu_\ell$ fully characterises them.
        The ADI pair step handles active-active coupling exactly.
      }
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.82, >=stealth, every node/.style={font=\scriptsize}]
        % Central voxel
        \draw[very thick, fill=activegreen!15, rounded corners=3pt]

\begin{frame}{Adaptive resolution: active vs.\ equilibrated voxels}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}

      \begin{block}{Active voxels (wavefront)}
        Full per-voxel CME via $\mathrm{expv}(dt, Q_k, p_k)$.\\
        Tridiagonal generator $Q_k$ --- $O(w)$ Krylov cost where
        $w \approx 20\text{--}50$ is the active support width.
      \end{block}

      \medskip
      \begin{block}{Equilibrated (equil) voxels}
        Distribution locked near a stable basin.\\
        \textbf{Boundary equil} (adjacent to active): $\mathrm{expv}$ at
        reduced Krylov dimension $m_{\rm eq}{=}8$.\\
        \textbf{Interior equil}: algebraic update only ---
        $\mu_k \leftarrow (k_b + \text{in})/({k_d + \text{out}})$,
        $p_k \leftarrow \mathrm{Pois}(\mu_k)$.\\
        No matrix exponential needed.
      \end{block}

      \medskip
      \begin{block}{ADI pair step (active-active pairs)}
        Adjacent active voxels evolved as an exact \textbf{2-voxel RDME}
        (no mean-field between them). Row/column sweeps alternate.
      \end{block}

\begin{frame}{The ADI pair V-cycle: exact 2-voxel RDME at the front}

\begin{columns}[T]
\column{0.47\textwidth}

\textbf{For each adjacent active pair $(v_1, v_2)$:}\\[0.4em]
\begin{enumerate}\setlength\itemsep{0.28em}
  \item Build sparse joint IC: $p_1 \otimes p_2$ (product state)
  \item \textcolor{slowred}{\textbf{Pre-smooth}}: evolve intra-pair diffusion
        $v_1 \leftrightarrow v_2$ for $\tau_{\rm pre} = 0.1\,\Delta t$
  \item \textcolor{mygray}{\textbf{Restrict}}: marginalise to total count\\
        $p(N) = \sum_{n_1+n_2=N} p(n_1,n_2)$ \quad [1-D vector]
  \item \textcolor{fastblue}{\textbf{Coarse solve}}: 1-D CME on $N$ with\\
        external boundary flux, for full $\Delta t$
  \item \textcolor{activegreen}{\textbf{Prolong}}: multi-level reconstruction\\
        of joint $p_{\rm new}(n_1,n_2)$ from $p_{\rm post}(N)$
  \item \textcolor{slowred}{\textbf{Post-smooth}}: intra-pair diffusion for $\tau_{\rm post}$
  \item Extract marginals $p_1^{\rm new},\, p_2^{\rm new}$
\end{enumerate}

\medskip
\begin{block}{ADI alternation}
  \textbf{Even steps}: horizontal pairs $(i,j)$--$(i,j{+}1)$.\\
  \textbf{Odd steps}: vertical pairs $(i,j)$--$(i{+}1,j)$.\\
  Alternating captures all 4 nearest-neighbour correlations.
\end{block}

\begin{frame}{Multi-level prolongation: three-level hierarchy}

\begin{columns}[T]
\column{0.43\textwidth}

\textbf{The restriction erases the split.}\\[0.3em]
Marginalising $p(n_1,n_2)$ to $p(N)$ loses all information about
\emph{how $N$ molecules are distributed across $v_1$ and $v_2$}.

\medskip
\textbf{Three-level hierarchy on the split axis:}
\begin{itemize}\setlength\itemsep{0.3em}
  \item \textbf{L0 (fine)}: $p(n_1 \mid N)$ exact conditional
  \item \textbf{L1 (mid)}: $p(\lfloor n_1/2\rfloor \mid N)$
        --- pair consecutive $n_1$ into bins of width 2
  \item \textbf{L2 (coarse)}: $p(N)$ total count only
\end{itemize}

\medskip
\begin{block}{Prolongation rule}
  \textbf{Existing $N$} (in support of $p_{\rm pre}$):\\
  \quad exact: $p_{\rm new}(n_1,n_2) = p_{\rm pre}(n_1,n_2)
  \times \dfrac{p_{\rm post}[N]}{p_{\rm pre}[N]}$\\[0.5em]
  \textbf{New $N$} (coarse solve added probability):\\
  \quad 1.\ propagate L1 fiber via stochastic $\mathcal{L}/\mathcal{R}$\\
  \quad 2.\ refine L1$\to$L0 from reference fiber
\end{block}

\begin{frame}{Stochastic lifting operators: physics-derived interpolation}

\begin{columns}[T]
\column{0.50\textwidth}

\textbf{Why L/R and not product-convolution?}

\medskip
Product-convolution uses marginals $m_1(n_1)\cdot m_2(N{-}n_1)$
 (a flat prior that ignores which reaction changed $N$).

\medskip
\begin{block}{Stochastic lifting $\mathcal{L}$: \small$N \to N+1$}
When a molecule is born, it goes to $v_1$ or $v_2$ with probabilities
determined by the actual propensities:
\[
  \alpha_1(n_1) = \frac{b_1(n_1)}{b_1(n_1)+b_2(n_2)}, \quad
  b_i(n) = \alpha_{\rm rxn}(n) + d\,\mu_i^{\rm ext}
\]
\[
  \mathcal{L}\bigl[p(\cdot\mid N)\bigr](n_1) =
  \alpha_1(n_1{-}1)\,p(n_1{-}1\mid N) +
  \alpha_2(n_1)\,p(n_1\mid N)
\]
\end{block}

\medskip
\begin{block}{Stochastic lowering $\mathcal{R}$: \small$N \to N-1$}
Conditioned on a death event, molecule leaves from $v_1$ or $v_2$:
\[
  \beta_1(n_1) = \frac{d_1(n_1)}{d_1(n_1)+d_2(n_2)}, \quad
  d_i(n) = (k_d + d\,|\partial i|)\,n
\]
\end{block}

\column{0.46\textwidth}
\vspace{0.2em}
\begin{center}

\begin{frame}{Extending to true spatial multigrid: three spatial scales}

\begin{columns}[T]
\column{0.46\textwidth}

The pair V-cycle captures 2-voxel correlations.
A \textbf{full spatial multigrid} adds coarser levels:

\medskip
\begin{block}{Spatial hierarchy (active wavefront)}
\small
\begin{tabular}{lll}
  \textbf{L0} & individual voxels & $w$ states\\
  \textbf{L1} & adjacent pairs    & $p(N_2)$, 1-D\\
  \textbf{L2} & $2{\times}2$ quads & $p(N_4)$, 1-D
\end{tabular}
\end{block}

\medskip
\textbf{V-cycle (time split: $\alpha_{\rm pre} + (1{-}2\alpha) + \alpha_{\rm post} = \Delta t$):}
\begin{enumerate}\setlength\itemsep{0.25em}
  \item Pre-smooth L0: per-voxel expv for $\alpha\Delta t$
  \item Restrict L0$\to$L1: $p_{12}(N_2) = p_1 * p_2$ \,(convolution)
  \item Restrict L1$\to$L2: $p_{1234}(N_4) = p_{12} * p_{34}$
  \item \textbf{Coarse solve L2}: 1-D CME for $(1{-}2\alpha)\Delta t$
  \item Correct L2$\to$L1: $p_{12}^{\rm new} \propto p_{12}\times E[C(N_4)\mid N_2]$
  \item Correct L1$\to$L0: $p_k^{\rm new} \propto p_k\times E[C(N_2)\mid n_k]$
  \item Post-smooth L0: per-voxel expv for $\alpha\Delta t$
\end{enumerate}

\begin{frame}{Summary}

  \begin{block}{1.\quad Per-voxel CME with mean-field coupling.}
    The full RDME is never assembled.  Diffusion is approximated by a
    Poisson in-flux at the neighbor mean --- exact for sharp (equil) voxels.
    Each voxel evolves independently via $\mathrm{expv}$ on a compact
    tridiagonal generator.
  \end{block}

  \bigskip
  \begin{block}{2.\quad ADI pair V-cycle: exact 2-voxel RDME at the front.}
    Adjacent active voxel pairs solved exactly (no mean-field between them).
    Row/column alternation (ADI) gives isotropic 2-D propagation.
    Pre/post-smooth equilibrate the intra-pair split; a 1-D coarse CME
    captures the total-count dynamics with external boundary flux.
  \end{block}

  \bigskip
  \begin{block}{3.\quad Multi-level stochastic prolongation.}
    Three-level hierarchy on the split axis.
    Existing $N$: exact multiplicative correction.
    New $N$: physics-derived $\mathcal{L}/\mathcal{R}$ operators propagate the
    conditional fiber.  Strictly better than product-convolution for bistable systems.
  \end{block}

  \bigskip
  \begin{block}{4.\quad True spatial multigrid: L0$\to$L1$\to$L2.}
    Pair V-cycle extended to a three-level spatial hierarchy:
    per-voxel ($w$ states) $\to$ pairs ($p(N_2)$) $\to$ $2{\times}2$ quads ($p(N_4)$).
    Coarse solve at L2 captures collective 4-voxel dynamics; multiplicative
    corrections propagate L2$\to$L1$\to$L0.  Time split ensures total evolution
    time exactly equals $\Delta t$.\\[0.2em]
    \emph{${\sim}40\%$ speedup on $200{\times}200$ BD wavefront.
    Correction field confirms L2 adds information at the front, zero in interior.}
  \end{block}

\end{frame}
