\begin{frame}{The Chemical Master Equation: exact stochastic distributions}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      A well-mixed system: integer counts
      $\mathbf{n} \in \mathbb{Z}_{\geq 0}^S$ evolving by stochastic reactions.

      \medskip
      The probability $p(\mathbf{n}, t)$ satisfies the \textbf{CME}:
      \[
        \dot{p} = A\,p,
      \]
      where $A$ encodes every reaction event as a transition rate.

      \bigskip
      \begin{alertblock}{SSA vs.\ CME}
        \textbf{SSA} samples individual trajectories.\\[0.3em]
        \textbf{CME} gives the \emph{exact} joint distribution
        $p(\mathbf{n},t)$ directly --- rare events, tails, and
        multimodal structure without ensemble noise.
      \end{alertblock}

      \bigskip
      {\small
        \textbf{FSP}: truncate $A$ to a finite support and solve $\dot p = Ap$
        numerically.  Expand support at the boundary; prune negligible states.
      }
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{Birth-death: $\emptyset \xrightarrow{\lambda} X \xrightarrow{\mu n} \emptyset$}

      \smallskip
      \begin{center}
      \begin{tikzpicture}[scale=0.88, every node/.style={font=\small},
                          >=stealth]
        \foreach \n in {0,1,2,3,4} {
          \draw[thick,fill=myblue!15,rounded corners=2pt]
            (\n*1.5,0) rectangle (\n*1.5+0.9,0.7);
          \node at (\n*1.5+0.45,0.35) {$\n$};
        }
        \node at (4*1.5+1.15,0.35) {$\cdots$};
        \foreach \n in {0,1,2,3} {
          \draw[->,fastblue,thick,bend left=30]

        }
      \end{tikzpicture}
      \end{center}

      \vspace{0.5em}
      Stationary distribution: $\operatorname{Poisson}(\lambda/\mu)$.

\begin{frame}{The RDME: reactions and diffusion across a spatial grid}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Spatial extension of the CME.}
      Divide a 1-D domain into $K$ voxels.
      Each voxel $k$ holds molecule counts $\mathbf{n}_k \in \mathbb{Z}_{\geq 0}^S$.

      \medskip
      Two types of events:
      \begin{itemize}\setlength\itemsep{0.4em}
        \item \textcolor{slowred}{\textbf{Reactions}} inside voxel $k$:
              rate $\alpha_r(\mathbf{n}_k)$, change $\mathbf{n}_k$.
        \item \textcolor{fastblue}{\textbf{Diffusion}} between neighbors:
              rate $d = D/h^2$, moves one molecule $k \leftrightarrow k{+}1$.
      \end{itemize}

      \medskip
      The joint probability $p(\mathbf{n}_1,\ldots,\mathbf{n}_K,t)$
      satisfies $\dot p = A p$ \textbf{(RDME)}.

      \medskip
      \begin{alertblock}{The computational bottleneck}
        The joint state space has ${\sim}n_{\max}^{SK}$ entries ---
        \textbf{exponential in $K$}.
        Even $K{=}8$, $S{=}1$ is intractable with a naive FSP.
      \end{alertblock}

      \medskip
      {\small The generator $A$ has \textbf{block-tridiagonal} structure
      (reactions on the diagonal, diffusion off-diagonal) ---
      this locality is the key to our approach.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.90, every node/.style={font=\small}]
        \foreach \k in {1,2,3,4,5,6} {
          \draw[thick,fill=myblue!12] (\k-1,0) rectangle (\k,1.0);
          \node at (\k-0.5, 0.50) {$\mathbf{n}_\k$};
        }
        \foreach \k in {1,2,3,4,5} {
          \draw[<->,fastblue,thick] (\k-0.15,1.18) -- (\k+0.15,1.18);
        }
        \node[fastblue, font=\scriptsize] at (3, 1.45) {diffusion $d = D/h^2$};
        \foreach \k in {1,2,3,4,5,6} {
          \draw[->,slowred,thick] (\k-0.5,-0.05) -- (\k-0.5,-0.38);
        }
        \node[slowred, font=\scriptsize] at (3, -0.58) {reactions $\alpha_r(\mathbf{n}_k)$};
        \draw[<->,thick,mygray] (0,-1.1) -- (6,-1.1)
          node[midway,below,font=\scriptsize]{$K$ voxels};
        \node[mygray,font=\scriptsize] at (3,-1.65)
          {state space: $n_{\max}^{SK}$ entries};
      \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}

\end{frame}

  \medskip
  \textcolor{slowred!80!black}{$D_{k,k+1}$}: diffusion $k \leftrightarrow k{+}1$
  \quad (sparse: shifts one count)

  \begin{block}{Key locality}
    Diffusion only couples \emph{adjacent} blocks.\\[0.4em]

    \textbf{This is the foundation of spatial coarsening}:\\

\begin{frame}{The Binomial is the exact diffusion equilibrium}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Fix total molecules} $\bar n = n_1 + n_2$ across two adjacent voxels.

      \smallskip
      Diffusion moves one molecule at a time, at rate $d$ per molecule:

      \begin{center}
      \begin{tikzpicture}[every node/.style={font=\small}, scale=0.88]
        \draw[thick,fill=coarsecolor!15] (0,0) rectangle (1.3,0.85);
        \draw[thick,fill=coarsecolor!15] (2.8,0) rectangle (4.1,0.85);
        \node at (0.65, 0.42) {$n_1$};
        \node at (3.45, 0.42) {$n_2$};
        \draw[->,fastblue,thick,bend left=22] (1.3,0.58)
          to node[above,font=\scriptsize]{rate $d\cdot n_1$} (2.8,0.58);
        \draw[->,slowred,thick,bend left=22]  (2.8,0.27)
          to node[below,font=\scriptsize]{rate $d\cdot n_2$} (1.3,0.27);
      \end{tikzpicture}
      \end{center}

      likely to be in either voxel. For $\bar n$ total:

        \boxed{P(n_1 \mid \bar n) = \Bin\!\bigl(\bar n,\,\tfrac{1}{2}\bigr)}
        \quad \text{(exact, any } d > 0\text{)}
      \]
      When the distribution is near this equilibrium,
      \textbf{$\bar n$ alone describes the two-voxel state}.
    \end{column}

\begin{frame}{Merging two adjacent voxels: restriction and prolongation}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Compress ($\calR$): exact:}
      \[ p^c(\bar n) = \!\!\sum_{n_1 + n_2 = \bar n}\!\! p(n_1, n_2). \]

      \textbf{Reconstruct ($\calP$): approximate:}
      \[ p(n_1, n_2) \approx p^c(\bar n)\cdot\pi(\bar n), \]
      where $\pi$ is the \textbf{within-pair distribution}.

      \begin{block}{State-space reduction}
        Per pair: $n_{\max}^{2S} \to n_{\max}^{S}$.\\
        All $K$ voxels: $n_{\max}^{SK} \to n_{\max}^{SK/2}$.
      \end{block}

      \smallskip
      {\small
        $\pi = \Bin(\bar n,\tfrac12)$: exact when diffusion $\gg$ reactions
        (within-pair distribution at equilibrium).\\[0.3em]
        \textbf{Birth-death:} this is essentially exact.\\
        \textbf{Bistable reactions:} requires a different $\pi$ --- see later.
      }
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{center}

\begin{frame}{The V-cycle: multigrid FSP}

\begin{columns}[T]
\column{0.44\textwidth}

\textbf{One V-cycle ($K \to K/2 \to K$):}\\[0.5em]
\begin{enumerate}\setlength\itemsep{0.35em}
  \item \textbf{Pre-smooth}: intra-pair diffusion for $\tau_{\rm pre}$
  \item \textbf{Restrict}: $\bar n_j = n_{2j-1}+n_{2j}$ \quad (exact)
  \item \textbf{Coarse FSP}: adaptive solve on $K/2$ voxels
  \item \textbf{Prolong}: reconstruct fine distribution
  \item \textbf{Prune}: drop states with $p(x) < \tau$
\end{enumerate}

\medskip
\textbf{Prolongation for covered states:}
\[
  p^{\rm new}(x) = \tilde p(x)\cdot\frac{\bar p^{\rm post}(\bar x)}{\bar p^{\rm pre}(\bar x)}
\]
{\small Multiplicative update --- exact for birth-death;
preserves bistable orientation for Schl\"{o}gl.}

\column{0.53\textwidth}
\vspace{0.3em}

\begin{frame}{Bistable reactions}

\begin{columns}[T]
\column{0.50\textwidth}
For birth-death, each voxel has one stable state.
Diffusion wins: the pair is always well-mixed.
Binomial coarsening is \emph{essentially exact}.

\vspace{0.8em}
\textbf{But what if reactions are bistable?}\\[0.4em]
The \textbf{Schl\"{o}gl model} has two stable states per voxel:
$n_\text{lo} \approx 31$ and $n_\text{hi} \approx 162$.

\vspace{0.4em}
\begin{center}
  \includegraphics[width=0.95\textwidth]{fig_talk_bistability}
\end{center}

\column{0.46\textwidth}
\vspace{0.3em}
\begin{block}{Spatial metastability}
Neighbors can occupy \emph{different} states.
A \textbf{spatial front} forms.
\end{block}

% ================================================================
%  9. Why Binomial fails
% ================================================================
\begin{frame}{Why Binomial prolongation fails for bistable fronts}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      At the spatial front, neighboring voxels occupy \emph{different} stable states.
      The joint distribution $P(n_1, n_2)$ is \textbf{bimodal}:
      mass at $(n_\text{lo}, n_\text{hi})$ \emph{and} $(n_\text{hi}, n_\text{lo})$.

      \bigskip
      \begin{alertblock}{The symmetry trap}
        Restriction maps both front orientations to the same scalar:
        \[
          \calR(n_\text{lo}, n_\text{hi}) \;=\; \bar n \;=\; \calR(n_\text{hi}, n_\text{lo}).
        \]
        \textbf{Front direction is erased.}\\[0.3em]
        Binomial prolongation places the reconstructed mass at the
        \textbf{unstable midpoint} $\approx 96$
        --- the system immediately leaves this state.
      \end{alertblock}

      \bigskip
      {\small\textbf{Fix:} multiplicative update (V-cycle step 4) rescales the
      \emph{existing} fine distribution without reconstructing it from scratch.
      Mass at $(n_{\rm lo},n_{\rm hi})$ stays exactly there.}
    \end{column}

    \begin{column}{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=1.0]
        \draw[->] (0,0) -- (4.3,0) node[right,font=\small]{$n_1$};
        \draw[->] (0,0) -- (0,4.3) node[above,font=\small]{$n_2$};
        % stable state labels on axes
        \foreach \x/\lbl in {0.55/$n_{\rm lo}$, 3.55/$n_{\rm hi}$}
          \node[below,font=\tiny,mygray] at (\x,-0.12) {\lbl};
        \foreach \y/\lbl in {0.55/$n_{\rm lo}$, 3.55/$n_{\rm hi}$}
          \node[left,font=\tiny,mygray]  at (-0.12,\y) {\lbl};
        % bimodal mass
        \fill[slowred!45, opacity=0.85] (0.55,3.55) circle (0.42);
        \fill[slowred!45, opacity=0.85] (3.55,0.55) circle (0.42);
        \node[white,font=\tiny\bfseries] at (0.55,3.55) {lo$|$hi};
        \node[white,font=\tiny\bfseries] at (3.55,0.55) {hi$|$lo};
        % restriction diagonal
        \draw[coarsegreen!70,dashed,thick] (0.1,3.9) -- (3.9,0.1);
        \node[coarsegreen!70!black,font=\scriptsize,rotate=-45] at (2.65,1.25)
          {restrict: $\bar n = n_\text{lo}+n_\text{hi}$};
        % Binomial places mass at midpoint
        \fill[intercolor,opacity=0.80] (2.05,2.05) circle (0.32);
        \draw[->,thick,intercolor!80!black] (1.2,3.1) -- (1.97,2.13);
        \node[intercolor!80!black,font=\scriptsize,above] at (1.0,3.15)
          {$\Bin(\bar n,\tfrac12)$};
        \node[intercolor!80!black,font=\tiny] at (2.05,2.05) {${\approx}96$};
      \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}

\end{frame}

% ================================================================
%  11. Summary
% ================================================================
\begin{frame}{Summary}

  \begin{block}{1.\quad Restriction + prolongation compresses the RDME state space.}
    Marginalizing adjacent-voxel pairs reduces the state space from
    $n_{\max}^{SK}$ to $n_{\max}^{SK/2}$ per V-cycle level.
    Binomial prolongation is exact in the fast-diffusion limit.\\[0.3em]
    \emph{K=8 birth-death wavefront solved via K=4 coarse FSP (30k states; direct: intractable).}
  \end{block}

  \bigskip
  \begin{block}{2.\quad Conditional prolongation handles bistable fronts.}
    Multiplicative update for covered states preserves within-pair asymmetry ---
    mass at $(n_{\rm lo},n_{\rm hi})$ is never symmetrised to the unstable midpoint.\\[0.3em]
    $L_1{\leq}2$ neighbor search correctly seeds new states at both reaction
    and diffusion boundaries, propagating the front across pairs.\\[0.3em]
    \emph{Schl\"{o}gl wavefront ($K{=}4$, controlled state-space growth) correctly captured.}
  \end{block}

\end{frame}
