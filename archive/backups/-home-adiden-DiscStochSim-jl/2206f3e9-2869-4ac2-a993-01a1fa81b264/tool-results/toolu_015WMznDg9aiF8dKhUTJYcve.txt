   600→\end{align*}
   601→The solution of the inhomogeneous ODE is given by,
   602→\begin{align}
   603→\boldsymbol{\varepsilon}(t_{n+1}) &= \int_{t_n}^{t_{n+1}} e^{(t_{n+1}-s)\mathbf{A}_{JJ}}\boldsymbol{f}(s) \, ds + e^{A_{JJ}}\varepsilon(t_n). \\
   604→\end{align}
   605→We set the initial condition $\varepsilon(t_n)$ to zero as we start both systems with same initial condition. Eliminating the initial condition and taking the $\ell_1$ norm both sides, we get:
   606→\begin{align}
   607→\|\boldsymbol{\varepsilon}(t_{n+1})\|_1 &\leq \int_{t_n}^{t_{n+1}} \underbrace{\|e^{(t_{n+1}-s)\mathbf{A}_{JJ}}\|_{1}}_{\leq 1 (\text{Corollary \ref{cor:truncated-contraction}})}\|\boldsymbol{f}(s)\|_1 \, ds \\
   608→&\leq \int_{t_n}^{t_{n+1}} \left(\|\mathbf{A}_{JJ'}\boldsymbol{p}_{J'}(s)\|_1  +\|\diag(\mathbf{1}^\top\mathbf{A}_{J'J})\widehat{\boldsymbol{p}}_J(s)\|_1 \right)  \; ds \\
   609→&= \int_{t_n}^{t_{n+1}} ( (\mathbf{1}^\top \mathbf{A}_{JJ'})\boldsymbol{p}_{J'}(s) + (\mathbf{1}^\top\mathbf{A}_{J'J})\widehat{\boldsymbol{p}}_J(s)) \;ds \\
   610→&= \int_{t_n}^{t_{n+1}} (\Phi_{\mathrm{in}}(s) + \Phi_{\mathrm{out}}(s)) \; ds.
   611→\end{align}
   612→
   613→\end{proof}
   614→
   615→While potentially conservative, this characterization provides a computationally tractable error estimate based on readily available flux quantities. We next analyze the accumulation of these local errors over multiple time steps.
   616→
   617→\subsubsection{Matrix Exponential Approximation Error}
   618→\label{subsec:time_stepping_error_analysis}
   619→
   620→In addition to state-space truncation, we approximate the matrix exponential $\exp(\widetilde{\mathbf{A}}_{JJ}\,\Delta t)$ numerically. 
   621→
   622→\begin{proposition}[Local time-stepping error]
   623→\label{prop:local_time_error}
   624→For a matrix exponential approximation with tolerance $\epsilon_{\mathrm{ODE}}$, the error in a single time step is bounded by
   625→\begin{equation}
   626→\|\exp(\widetilde{\mathbf{A}}_{JJ}\,\Delta t)\,\boldsymbol{v} - \mathrm{Approx}(\widetilde{\mathbf{A}}_{JJ}, \Delta t)\,\boldsymbol{v}\|_1 \le \epsilon_{\mathrm{ODE}}\,\|\boldsymbol{v}\|_1.
   627→\end{equation}
   628→Modern Krylov subspace implementations such as EXPOKIT \cite{Sidje1998} provide adaptive mechanisms to control this error by adjusting the Krylov subspace dimension based on the specified tolerance.
   629→\end{proposition}
   630→
   631→\subsection{Global Error}
   632→
   633→\begin{corollary}[Global error bound]
   634→\label{cor:global-error}
   635→Let $e_n := \|\boldsymbol{p}(t_n) - \widetilde{\boldsymbol{p}}_n\|_1$ denote the global error, where $\widetilde{\boldsymbol{p}}_n$ is the numerical solution on $J$ extended with zeros. For $N$ time steps with $e_0 = 0$,
   636→\begin{equation}
   637→e_N \leq \sum_{n=0}^{N-1} (\varepsilon_n + \tau_n),
   638→\end{equation}
   639→where $\varepsilon_n := \int_{t_n}^{t_{n+1}} (\Phi_{\mathrm{in}}(s) + \Phi_{\mathrm{out}}(s)) \, ds$ is the local model error and $\tau_n := \|\widehat{\boldsymbol{p}}_{n+1} - \widetilde{\boldsymbol{p}}_{n+1}\|_1$ is the time-stepping error.
   640→\end{corollary}
   641→
   642→\begin{proof}
   643→We track three sources of discrepancy:
   644→\begin{itemize}
   645→\item $\boldsymbol{p}(t_n)$: true solution on full state space
   646→\item $\widehat{\boldsymbol{p}}_n$: exact solution of compressed dynamics on $J_n$ 
   647→\item $\widetilde{\boldsymbol{p}}_n$: numerical approximation on $J_n$
   648→\end{itemize}
   649→
   650→By the triangle inequality:
   651→\begin{equation}
   652→e_n = \|\boldsymbol{p}(t_n) - \widetilde{\boldsymbol{p}}_n\|_1 \leq \|\boldsymbol{p}(t_n) - \widehat{\boldsymbol{p}}_n\|_1 + \|\widehat{\boldsymbol{p}}_n - \widetilde{\boldsymbol{p}}_n\|_1.
   653→\end{equation}
   654→
   655→For the first term (model error), Theorem~\ref{thm:local-model-error} and the contraction property give:
   656→\begin{align}
   657→\|\boldsymbol{p}(t_{n+1}) - \widehat{\boldsymbol{p}}_{n+1}\|_1 
   658→&\leq \|\boldsymbol{p}(t_n) - \widehat{\boldsymbol{p}}_n\|_1 + \varepsilon_n.
   659→\end{align}
   660→
   661→For the second term (numerical error), Proposition~\ref{prop:local_time_error} gives:
   662→\begin{equation}
   663→\|\widehat{\boldsymbol{p}}_{n+1} - \widetilde{\boldsymbol{p}}_{n+1}\|_1 \leq \|\widehat{\boldsymbol{p}}_n - \widetilde{\boldsymbol{p}}_n\|_1 + \tau_n.
   664→\end{equation}
   665→
   666→Combining these yields $e_{n+1} \leq e_n + \varepsilon_n + \tau_n$, and recursively expanding with with $e_0 = 0$ gives the result.
   667→\end{proof}
   668→
   669→Having established the local error bound in Theorem~\ref{thm:local-model-error}, we now connect the theoretical error quantities to the computational tolerances that control the adaptive algorithm. The following corollaries show how the user-specified tolerances: $\alpha$ (probability mass threshold for pruning): $\epsilon_{\mathrm{flux}}$ (flux preservation threshold), and $\epsilon_{\Delta t}$ (time step error tolerance), translate into error bounds.
   670→
   671→\begin{corollary}[Combined flux and probability error control]
   672→\label{cor:local-error-pruning}
   673→Under flux-based pruning with probability tolerance $\alpha$ and flux tolerance $\epsilon_{\mathrm{flux}}$, the local error over $[t_n, t_{n+1}]$ satisfies, to leading order in $\Delta t_n$,
   674→\begin{equation}
   675→\|\boldsymbol{\varepsilon}(t_{n+1})\|_1 \lesssim (\epsilon_{\mathrm{flux}} \cdot \Phi_{\mathrm{total}}(t_n) + \alpha \cdot w_{\max}) \cdot \Delta t_n,
   676→\end{equation}
   677→where $w_{\max} = \max_{x \in \mathcal{S}_{\mathrm{prune}}} w(x)$ is the maximum exit rate among pruned states. Crucially, $w_{\max}$ is computable at pruning time: Algorithm~\ref{alg:flux-prune} evaluates $w(\boldsymbol{x})$ for every state in the active set $\mathcal{S}$ (Step~7) \emph{before} any states are removed, so $w_{\max}$ is obtained as a byproduct by recording $\max_{\boldsymbol{x} \in \mathcal{C}} w(\boldsymbol{x})$ over the candidate set $\mathcal{C} \supseteq \mathcal{S}_{\mathrm{prune}}$.
   678→\end{corollary}
   679→\begin{proof}
   680→By Theorem~\ref{thm:local-model-error}, the local model error satisfies
   681→\begin{equation}
   682→\|\boldsymbol{\varepsilon}(t_{n+1})\|_1 \leq \int_{t_n}^{t_{n+1}} (\Phi_{\mathrm{in}}(s) + \Phi_{\mathrm{out}}(s)) \, ds \approx (\Phi_{\mathrm{in}}(t_n) + \Phi_{\mathrm{out}}(t_n)) \cdot \Delta t_n
   683→\end{equation}
   684→for small $\Delta t_n$. Since $J'_n$ carries negligible probability after pruning, $\Phi_{\mathrm{in}}(t_n) \approx 0$ and the error is dominated by $\Phi_{\mathrm{out}}(t_n)$.
   685→
   686→By Algorithm~\ref{alg:flux-prune}, the pruned set $\mathcal{S}_{\mathrm{prune}} = \mathcal{C} \setminus \mathcal{P}$ consists of states with total probability mass at most $\alpha$ (from quantile selection) and flux below $\epsilon_{\mathrm{flux}} \cdot \Phi_{\mathrm{total}}$ (failed flux protection). The flux contribution from pruned states satisfies
   687→\begin{equation}
   688→\sum_{x \in \mathcal{S}_{\mathrm{prune}}} p(x, t_n) \cdot w(x) \leq w_{\max} \sum_{x \in \mathcal{S}_{\mathrm{prune}}} p(x, t_n) \leq \alpha \cdot w_{\max}.
   689→\end{equation}
   690→States retained by flux protection contribute at most $\epsilon_{\mathrm{flux}} \cdot \Phi_{\mathrm{total}}(t_n)$ to boundary flux by construction. Combining these contributions yields $\Phi_{\mathrm{out}}(t_n) \leq \epsilon_{\mathrm{flux}} \cdot \Phi_{\mathrm{total}}(t_n) + \alpha \cdot w_{\max}$, which gives the stated bound.
   691→\end{proof}
   692→
   693→This bound separates the two error sources controlled by the algorithm. The first term captures flux leakage from connectivity-critical states retained at the protection threshold, while the second term bounds the contribution from low-probability states removed during quantile-based pruning. The bound is conservative in two respects: it assumes worst-case alignment where all pruned states have the maximum exit rate $w_{\max}$, and it treats flux-protected states as contributing their full threshold flux to the boundary. In practice, the actual error is typically smaller since pruned states have heterogeneous exit rates and protected states often lie in the interior of the active set rather than at the truncation boundary.
   694→
   695→\begin{corollary}[Adaptive time step]
   696→\label{cor:flux-timestep}
   697→Let $\kappa_n := \lambda_{\max}(J_n,t_n)/\Phi_{\mathrm{total}}(J_n,t_n) \geq 1$ denote the \emph{spatial stiffness ratio}, where $\lambda_{\max}(J_n,t_n) := \max_{\boldsymbol{x}\in J_n} w(\boldsymbol{x})$ and $\Phi_{\mathrm{total}}(J_n,t_n) = \sum_{\boldsymbol{x}\in J_n}p(\boldsymbol{x},t_n)\,w(\boldsymbol{x})$.
   698→Two time-step rules control the local model error within tolerance $\epsilon_{\Delta t}>0$:
   699→\begin{enumerate}[(i)]
   700→\item \textbf{Total-flux rule (recommended):}
   701→\begin{equation}
   702→\label{eq:dt-total}
   703→\Delta t_n = \frac{\epsilon_{\Delta t}}{\Phi_{\mathrm{total}}(J_n, t_n)}.
   704→\end{equation}
   705→This guarantees $\|\boldsymbol{\varepsilon}(t_{n+1})\|_1 \lesssim \epsilon_{\Delta t}$ to leading order in $\Delta t_n$, with expansion factor $\mu_n := \lambda_{\max}\,\Delta t_n = \kappa_n\,\epsilon_{\Delta t}$.
   706→\item \textbf{Maximum-flux rule (concentrated distributions only):} Let $\Phi_{\max}(J_n,t_n) := \max_{\boldsymbol{x}\in J_n}\{p(\boldsymbol{x},t_n)\,w(\boldsymbol{x})\}$ and define the \emph{concentration ratio} $\eta_n := \Phi_{\mathrm{total}}/\Phi_{\max} \geq 1$.
   707→\begin{equation}
   708→\label{eq:dt-max}
   709→\Delta t_n = \frac{\epsilon_{\Delta t}}{\Phi_{\max}(J_n, t_n)}.
   710→\end{equation}
   711→Since $\Phi_{\max} \leq \Phi_{\mathrm{total}}$, this step is larger than rule~(i) by factor $\eta_n$.
   712→The error satisfies $\|\boldsymbol{\varepsilon}(t_{n+1})\|_1 \lesssim \eta_n\,\epsilon_{\Delta t}$ to leading order in $\Delta t_n$, and the expansion factor is $\mu_n = (\lambda_{\max}/\Phi_{\max})\,\epsilon_{\Delta t}$.
   713→The condition $\mu_n \leq 1$ (depth-one sufficiency, Corollary~\ref{cor:expansion-depth}) holds if and only if $\epsilon_{\Delta t} \leq \Phi_{\max}/\lambda_{\max}$.
   714→When the distribution is concentrated ($\eta_n \approx 1$, e.g., systems with conservation laws that confine probability to a narrow stoichiometric manifold), both rules yield comparable steps and the error overhead is negligible.
   715→\textbf{Caution:} When $\eta_n \gg 1$ (spread distributions), the max-flux step $\Delta t_n^{\max} = \eta_n\,\Delta t_n^{\mathrm{tot}}$ is much larger than the total-flux step, and the leading-order bound grows as $\eta_n\,\epsilon_{\Delta t}$.
   716→For oscillatory systems (e.g., the Oregonator), the distribution spreads widely during autocatalytic bursts---probability is distributed across $O(|\mathcal{S}|)$ states, so $\eta_n \gg 1$---and the max-flux rule takes steps that the depth-one boundary expansion cannot track, causing the wavefront to escape the active set and the solution to collapse.
   717→The total-flux rule~(i) is free of this failure mode and is the recommended default.
   718→\end{enumerate}
   719→\end{corollary}
   720→
   721→\begin{proof}
   722→By Theorem~\ref{thm:local-model-error}, approximating boundary fluxes as constant over $[t_n,t_{n+1}]$:
   723→\begin{equation}
   724→\|\boldsymbol{\varepsilon}(t_{n+1})\|_1 \approx [\Phi_{\mathrm{in}}(t_n) + \Phi_{\mathrm{out}}(t_n)]\,\Delta t_n.
   725→\end{equation}
   726→For the outflow, since the rate of leaving any state $\boldsymbol{x} \in J$ into the complement $J'$ is at most $w(\boldsymbol{x})$, i.e.,\ $\sum_{\boldsymbol{y} \in J'} \mathbf{A}_{\boldsymbol{y}\boldsymbol{x}} \leq w(\boldsymbol{x})$:
   727→\begin{equation}
   728→\Phi_{\mathrm{out}}(t_n) = \sum_{\boldsymbol{x} \in J}\!\Big(\sum_{\boldsymbol{y} \in J'}\!\mathbf{A}_{\boldsymbol{y}\boldsymbol{x}}\Big)p(\boldsymbol{x},t_n) \leq \sum_{\boldsymbol{x}\in J}w(\boldsymbol{x})\,p(\boldsymbol{x},t_n) = \Phi_{\mathrm{total}}(J,t_n).
   729→\end{equation}
   730→For the inflow, all states in $J'\setminus\mathcal{S}_{\mathrm{prune}}$ carry $p(\boldsymbol{x},t_n)=0$ (never in the active set), and $\sum_{\boldsymbol{x}\in\mathcal{S}_{\mathrm{prune}}}p(\boldsymbol{x},t_n)\leq\alpha$ by the quantile selection of Algorithm~\ref{alg:flux-prune}, so $\Phi_{\mathrm{in}}(t_n)\leq w_{\max}(J')\cdot\alpha$.  Since $\alpha$ is chosen small this is negligible, giving the key bound:
   731→\begin{equation}
   732→\label{eq:key-flux-bound}
   733→\|\boldsymbol{\varepsilon}(t_{n+1})\|_1 \lesssim \Phi_{\mathrm{total}}(J_n,t_n)\,\Delta t_n.
   734→\end{equation}
   735→
   736→\textit{Case~(i):} Substituting $\Delta t_n = \epsilon_{\Delta t}/\Phi_{\mathrm{total}}$ into~\eqref{eq:key-flux-bound} yields $\|\boldsymbol{\varepsilon}(t_{n+1})\|_1 \leq \epsilon_{\Delta t}$.  The expansion factor is $\mu_n = \lambda_{\max}\,\Delta t_n = \lambda_{\max}\,\epsilon_{\Delta t}/\Phi_{\mathrm{total}} = \kappa_n\,\epsilon_{\Delta t}$.
   737→
   738→\textit{Case~(ii):} Since $\Phi_{\max}\leq\Phi_{\mathrm{total}}$, substituting $\Delta t_n = \epsilon_{\Delta t}/\Phi_{\max}$ into~\eqref{eq:key-flux-bound} gives
   739→\[
   740→\|\boldsymbol{\varepsilon}(t_{n+1})\|_1 \lesssim \Phi_{\mathrm{total}}\cdot\frac{\epsilon_{\Delta t}}{\Phi_{\max}} = \eta_n\,\epsilon_{\Delta t}.
   741→\]
   742→The expansion factor is $\mu_n = \lambda_{\max}\,\epsilon_{\Delta t}/\Phi_{\max}$, and $\mu_n\leq 1$ is equivalent to $\epsilon_{\Delta t}\leq\Phi_{\max}/\lambda_{\max}$.
   743→\end{proof}
   744→
   745→\begin{remark}[Maximum-flux rule as a practical speedup]
   746→\label{rem:max-flux-speedup}
   747→The maximum-flux rule~(ii) can serve as an efficient alternative to the total-flux
   748→rule~(i) for stiff systems.  Because $\Delta t_n^{\max} = \eta_n\,\Delta t_n^{\mathrm{tot}}$
   749→and the iteration count scales as $N \sim t_f/\Delta t$, using the max-flux rule with
   750→tolerance $\epsilon_{\Delta t}^{\max}$ is equivalent in step count to the total-flux rule
   751→with $\epsilon_{\Delta t}^{\mathrm{tot}} = \epsilon_{\Delta t}^{\max}\cdot\eta_n$.
   752→Choosing $\epsilon_{\Delta t}^{\max} = \epsilon_{\Delta t}^{\mathrm{tot}}/\eta_n$ therefore
   753→recovers the same iteration count while preserving the per-step error bound $\eta_n\,\epsilon_{\Delta t}^{\max} = \epsilon_{\Delta t}^{\mathrm{tot}}$.
   754→In practice, the concentration ratio $\eta_n$ is not known in advance; however, for
   755→the Robertson benchmark the empirical median $\eta_n \approx 107$, so setting
   756→$\epsilon_{\Delta t}^{\max} = 0.01$ with the max-flux rule and
   757→$\epsilon_{\Delta t}^{\mathrm{tot}} = 1.0$ with the total-flux rule produce
   758→virtually identical step counts (41\,799 vs.\ 58\,543 for $t\in[0,100]$) and identical
   759→accuracy ($\langle A\rangle = 6146$ in both cases).
   760→The total-flux rule with $\epsilon_{\Delta t}=1$ (approximately one reaction per step)
   761→is therefore the recommended universal default, as it requires no system-dependent
   762→tuning of $\epsilon_{\Delta t}$.
   763→\end{remark}
   764→
   765→\begin{remark}[Max-propensity cap and depth-one guarantee]
   766→\label{rem:max-propensity-cap}
   767→The expansion factor $\mu_n$ governs the expansion depth required by Corollary~\ref{cor:expansion-depth}: under the total-flux rule, $\mu_n = \kappa_n\,\epsilon_{\Delta t}$ exceeds one when $\kappa_n > 1/\epsilon_{\Delta t}$ (high spatial stiffness).  To guarantee depth-one sufficiency ($\mu_n\leq 1$) for arbitrary stiffness, apply the \emph{max-propensity cap}:
   768→\begin{equation}
   769→\label{eq:dt-cap}
   770→\Delta t_n = \min\!\left(\frac{\epsilon_{\Delta t}}{\Phi_{\mathrm{total}}(J_n,t_n)},\;\frac{1}{\lambda_{\max}(J_n,t_n)}\right).
   771→\end{equation}
   772→This ensures $\mu_n = \lambda_{\max}\,\Delta t_n \leq 1$ unconditionally.
   773→When $\kappa_n\,\epsilon_{\Delta t}\leq 1$ the cap is inactive and rule~(i) applies with error $\leq\epsilon_{\Delta t}$.
   774→When $\kappa_n\,\epsilon_{\Delta t}>1$ the cap activates with $\Delta t_n = 1/\lambda_{\max}$, giving
   775→\[
   776→\|\boldsymbol{\varepsilon}(t_{n+1})\|_1 \lesssim \Phi_{\mathrm{total}}/\lambda_{\max} = 1/\kappa_n \leq \epsilon_{\Delta t}:
   777→\]
   778→the local error is actually \emph{smaller} than the prescribed tolerance while expansion depth remains one.  The cap is most valuable for highly stiff systems (large $\kappa_n$) where maintaining $d=1$ is critical for computational tractability.
   779→\end{remark}
   780→
   781→\begin{corollary}[Expansion depth for boundary control]
   782→\label{cor:expansion-depth}
   783→Let $\lambda_{\max}(J_n, t_n) := \max_{\boldsymbol{x} \in J_n,\, k} \alpha_k(\boldsymbol{x})$ denote the maximum propensity in the active set at step $n$.
   784→After expanding $J_n$ by graph depth $d$ (so that every state in the complement $J'$ requires at least $d+1$ consecutive reaction steps to reach from any state in $J_n$), the expansion-induced boundary outflux satisfies
   785→\begin{equation}
   786→\label{eq:expansion-outflux-bound}
   787→\int_{t_n}^{t_{n+1}} \Phi_{\mathrm{out}}(s)\, ds
   788→\;\leq\;
   789→\lambda_{\max} \,\Delta t_n \cdot \mathrm{P}\!\bigl(\mathrm{Poisson}(\lambda_{\max}\,\Delta t_n) \geq d\bigr).
   790→\end{equation}
   791→To ensure this contribution to the local model error (Theorem~\ref{thm:local-model-error}) is at most $\varepsilon_{\mathrm{exp}} > 0$, choose
   792→\begin{equation}
   793→\label{eq:expansion-depth}
   794→d_{\min} = \Bigl\lceil \mu + z_\varepsilon\,\sqrt{\mu} \Bigr\rceil,
   795→\qquad \mu := \lambda_{\max}\,\Delta t_n,
   796→\end{equation}
   797→where $z_\varepsilon = \Phi^{-1}(1 - \varepsilon_{\mathrm{exp}}/\mu)$ and $\Phi$ is the standard normal CDF.
   798→Substituting the adaptive step $\Delta t_n = \varepsilon_{\Delta t}/\Phi_{\mathrm{total}}$ gives the depth in terms of the \emph{spatial stiffness ratio} $\kappa_n := \lambda_{\max}(J_n,t_n)/\Phi_{\mathrm{total}}(J_n,t_n) \geq 1$:
   799→\begin{equation}
   800→\label{eq:expansion-depth-stiffness}
   801→d_{\min} = \Bigl\lceil \kappa_n\,\varepsilon_{\Delta t} + z_\varepsilon\,\sqrt{\kappa_n\,\varepsilon_{\Delta t}} \Bigr\rceil.
   802→\end{equation}
   803→\end{corollary}
   804→
   805→\begin{proof}
   806→After expanding $J_n$ by depth $d$, every state in $J'$ lies at graph distance $\geq d+1$ from any state in $J_n$.
   807→Since probability mass in $J'$ is zero at time $t_n$ (all mass resides in $J_n$), the outflux $\Phi_{\mathrm{out}}$ is driven entirely by mass that has diffused from the interior of $J_n$ to its boundary during $[t_n, t_n+s]$.
   808→Because each reaction fires at rate $\leq \lambda_{\max}$, the number of reactions along any path in time $s$ is stochastically dominated by $\mathrm{Poisson}(\lambda_{\max}\,s)$.
   809→The probability mass reaching graph distance $d$ from $J_n$ (hence capable of flowing into $J'$) by time $t_n + s$ is therefore at most $\mathrm{P}(\mathrm{Poisson}(\lambda_{\max}\,s) \geq d)$.
   810→Hence $\Phi_{\mathrm{out}}(t_n+s) \leq \lambda_{\max} \cdot \mathrm{P}(\mathrm{Poisson}(\lambda_{\max}\,s) \geq d)$, and integrating using the monotonicity of the Poisson survivor in $s$:
   811→\[
   812→\int_{t_n}^{t_{n+1}} \Phi_{\mathrm{out}}(s)\,ds
   813→\;\leq\;
   814→\lambda_{\max}\,\Delta t_n \cdot \mathrm{P}\!\bigl(\mathrm{Poisson}(\lambda_{\max}\,\Delta t_n) \geq d\bigr).
   815→\]
   816→Setting $\mu = \lambda_{\max}\,\Delta t_n$ and applying the CLT approximation
   817→$\mathrm{P}(\mathrm{Poisson}(\mu) \geq d) \approx 1 - \Phi((d-\mu)/\sqrt{\mu})$,
   818→requiring $\mu \cdot \mathrm{P}(\mathrm{Poisson}(\mu) \geq d) \leq \varepsilon_{\mathrm{exp}}$ yields~\eqref{eq:expansion-depth}.
   819→Substituting $\Delta t_n = \varepsilon_{\Delta t}/\Phi_{\mathrm{total}}$ gives $\mu = \kappa_n\,\varepsilon_{\Delta t}$ and hence~\eqref{eq:expansion-depth-stiffness}.
   820→\end{proof}
   821→
   822→\begin{remark}[Expansion strategy and the depth-one regime]
   823→\label{rem:expansion-strategy}
   824→Corollary~\ref{cor:expansion-depth} shows that depth $d=1$ suffices whenever $\mu_n\leq 1$.  This is achieved by the max-propensity cap (Remark~\ref{rem:max-propensity-cap}) unconditionally, or by the maximum-flux rule when $\epsilon_{\Delta t}\leq\Phi_{\max}/\lambda_{\max}$.  Within this depth-one regime two practical expansion strategies are available:
   825→\begin{itemize}
   826→  \item \textbf{Stoichiometric expansion} adds every state reachable by a single reaction step from any active state, covering all $R$ reaction channels simultaneously.  This is the conservative choice, appropriate when $\kappa_n>1$ (multiple fast channels) or when the dominant reaction direction is not known \textit{a priori}.
   827→  \item \textbf{SSA expansion} samples one next-reaction direction per active state according to the propensity distribution, adding only the most probable single-step neighbor.  When $\kappa_n\approx 1$ (a single reaction channel dominates locally), SSA implements $d=1$ with substantially lower state-space growth: stoichiometric expansion inflates $|S_n|$ by up to factor $R$ per step, while SSA adds exactly one new state per active state.
   828→\end{itemize}
   829→In systems where $\kappa_n$ varies across the state space---one dominant channel locally but multiple channels relevant globally---stoichiometric expansion offers safety without requiring prior knowledge of which channel will dominate.
   830→\end{remark}
   831→
   832→\begin{remark}[Consistent parameter selection via spatial stiffness]
   833→\label{rem:consistent-params}
   834→Corollaries~\ref{cor:local-error-pruning}, \ref{cor:flux-timestep}, and~\ref{cor:expansion-depth} and Remarks~\ref{rem:max-propensity-cap} and~\ref{rem:expansion-strategy} provide a self-consistent framework for parameter selection, governed by the spatial stiffness ratio $\kappa=\lambda_{\max}/\Phi_{\mathrm{total}}$:
   835→\begin{itemize}
   836→  \item \textbf{Time step} (three options in increasing aggressiveness):
   837→    total-flux $\Delta t = \varepsilon_{\Delta t}/\Phi_{\mathrm{total}}$ (leading-order error $\lesssim\varepsilon_{\Delta t}$, $\mu=\kappa\,\varepsilon_{\Delta t}$; \textbf{recommended});
   838→    max-propensity cap $\Delta t = \min(\varepsilon_{\Delta t}/\Phi_{\mathrm{total}},\,1/\lambda_{\max})$ (error $\lesssim\min(\varepsilon_{\Delta t},1/\kappa)$, always $\mu\leq 1$);
   839→    maximum-flux $\Delta t = \varepsilon_{\Delta t}/\Phi_{\max}$ (error $\lesssim\eta\,\varepsilon_{\Delta t}$, $\mu\leq 1$ when $\varepsilon_{\Delta t}\leq\Phi_{\max}/\lambda_{\max}$; valid only when $\eta_n\approx 1$, otherwise risks solution collapse for spread distributions)
   840→    \hfill (Corollary~\ref{cor:flux-timestep}, Remark~\ref{rem:max-propensity-cap}),
   841→  \item \textbf{Expansion depth and strategy:} use $d_{\min}=\lceil\kappa\,\varepsilon_{\Delta t}\rceil$ with stoichiometric expansion under the total-flux rule; reduce to $d_{\min}=1$ via the cap or maximum-flux rule, choosing SSA expansion when $\kappa\approx 1$ (one dominant channel) and stoichiometric expansion when $\kappa>1$ \hfill (Corollary~\ref{cor:expansion-depth}, Remark~\ref{rem:expansion-strategy}),
   842→  \item \textbf{Flux tolerance:} $\varepsilon_{\mathrm{flux}} \ll \varepsilon_{\Delta t}$, e.g., $\varepsilon_{\mathrm{flux}} = \varepsilon_{\Delta t}/w_{\max}$ \hfill (Corollary~\ref{cor:local-error-pruning}).
   843→\end{itemize}
   844→In the bottleneck example of Section~\ref{sec:results}, $\kappa\approx k_2/k_1=10^5$ early in the simulation: the total-flux rule alone would require $d_{\min}\approx 10^5$, but the max-propensity cap enforces $\Delta t\leq 1/\lambda_{\max}=1/k_2$, reducing $d_{\min}$ to 1 while delivering local error $1/\kappa\ll\varepsilon_{\Delta t}$.
   845→The maximum-flux rule is advantageous for spatially stiff systems with concentrated distributions ($\kappa\gg 1$, $\eta\approx 1$): it takes larger steps than the total-flux rule while automatically maintaining $d=1$ when $\varepsilon_{\Delta t}\leq\Phi_{\max}/\lambda_{\max}$.
   846→
   847→\end{remark}

   865→\begin{remark}[Practical parameter selection]
   866→\label{rem:param-guidance}
   867→Table~\ref{tab:param_guidance} summarises concrete recommendations complementing the theoretical framework of Remark~\ref{rem:consistent-params}.
   868→Three rules of thumb emerge.
   869→
   870→\textbf{$\varepsilon_{\Delta t}$:} Choose $\varepsilon_{\Delta t}\leq 1/\kappa$ so that depth-one expansion suffices (Corollary~\ref{cor:expansion-depth}); for $\kappa\approx 1$ this permits $\varepsilon_{\Delta t}$ up to~1.
   871→Reduce $\varepsilon_{\Delta t}$ (to $0.01$--$0.1$) when a long integration horizon demands high accuracy, since local errors accumulate additively (Corollary~\ref{cor:global-error}).
   872→
   873→\textbf{$\alpha$:} For systems with connectivity bottlenecks, $\alpha$ can be set aggressively ($0.7$--$0.9$) because the flux criterion rescues any misclassified connector states; the quantile filter then removes only the genuinely irrelevant probability tail.
   874→Without bottlenecks, moderate values ($0.1$--$0.3$) avoid over-pruning.
   875→
   876→\textbf{$\varepsilon_{\mathrm{flux}}$:} Set to zero for oscillatory or well-mixed systems where the active set is simply connected and no low-probability connector states need protection (a positive value would retain unnecessary states near the expanding boundary).
   877→For bottleneck systems, match $\varepsilon_{\mathrm{flux}}$ to the expected relative flux of the connector state, $\varepsilon_{\mathrm{flux}}\sim p(s_b)\,w(s_b)/\Phi_{\mathrm{total}}$; in practice $10^{-12}$--$10^{-6}$ covers most cases.
   878→\end{remark}

   882→\label{sec:results}
   883→
   884→
   885→In this section we demonstrate the flux-aware FSP method on four benchmark systems from the stochastic chemical kinetics literature. We first present a detailed analysis on a toy bottleneck reaction system that validates the method's connectivity-preserving properties and error bounds along with a deterministic error analysis. The toggle switch tests spatial adaptivity for bimodal distributions. The Robertson system tests long-time integration with extreme temporal stiffness spanning nine orders of magnitude in rate constants. The Oregonator tests adaptive time-stepping for sustained oscillatory dynamics. Finally, we experimentally study the matrix construction algorithm's efficiency. All experiments were conducted on an M1 MacBook Pro (M1 Pro chip, 16 GB RAM). The adaptive FSP algorithm was implemented in Julia, using Catalyst.jl~\cite{catalyst2021} for reaction network modeling and Expokit.jl for Krylov-based matrix exponential computations via EXPOKIT's routines~\cite{expokit2021}.

  1053→\subsection{Oscillatory Dynamics in the Oregonator Model}
  1054→
  1055→We examine the Oregonator model, a reduced representation of the Belousov-Zhabotinsky oscillating chemical reaction developed by Field and Noyes \cite{field_noyes}. This three-species system captures the essential autocatalytic and inhibitory feedback mechanisms that generate sustained chemical oscillations. The Oregonator serves as a stringent test for numerical methods due to its combination of multiple timescales, relaxation oscillations, and sensitivity to parameter variations. The model consists of five elementary reactions involving three intermediate species: $X$ (HBrO$_2$), $Y$ (Br$^-$), and $Z$ (Ce$^{4+}$). The complete reaction network is presented in Table \ref{table:oregonator}. 
  1056→
  1057→The reaction network exhibits three distinct dynamical phases. The autocatalytic reaction (R3) drives explosive growth of $X$ when inhibitor $Y$ is depleted. The quadratic termination (R4) prevents unbounded growth at high $X$ concentrations. The recovery reaction (R5) slowly regenerates $Y$ from $Z$, resetting the cycle. Following Gillespie's methodology \cite{Gillespie1977}, we determine rate constants by analyzing the deterministic steady state. At equilibrium, the net flux for each species vanishes:
  1058→\begin{align}
  1059→\frac{dN_X}{dt} &= k_1 N_Y - k_2 N_X N_Y + k_3 N_X - 2k_4 N_X^2 = 0, \\
  1060→\frac{dN_Y}{dt} &= -k_1 N_Y - k_2 N_X N_Y + k_5 N_Z = 0, \\
  1061→\frac{dN_Z}{dt} &= k_3 N_X - k_5 N_Z = 0.
  1062→\end{align}
  1063→We specify the steady-state populations as $(N_X^*, N_Y^*, N_Z^*) = (500, 1000, 2000)$ molecules, placing the system in the oscillatory regime. From the third equation, $k_3 N_X^* = k_5 N_Z^*$, which implies $\mu_3 = \mu_5$, where $\mu_i = k_i N_i^*$ denotes the steady-state flux through reaction $i$. Defining the characteristic fluxes $\mu_1^* = k_1 N_Y^* = 2000$ s$^{-1}$ and $\mu_2^* = k_2 N_X^* N_Y^* = 50000$ s$^{-1}$, we obtain the rate constants summarized in Table~\ref{table:oregonator}.

  1101→\subsection{Stiff Dynamics in the Robertson Autocatalytic System}
  1102→
  1103→The Robertson system provides a severe test of numerical methods through its extreme stiffness, with reaction rates spanning nine orders of magnitude. Originally formulated by Robertson \cite{Robertson1967} for testing ODE solvers, this autocatalytic network has become a standard benchmark for stochastic simulation algorithms. The system involves three species undergoing the reactions shown in Table \ref{table:robertson}, with rate constants $k_1 = 0.04$ s$^{-1}$, $k_2 = 3 \times 10^7$ molecule$^{-1}$s$^{-1}$, and $k_3 = 10^4$ molecule$^{-1}$s$^{-1}$. This yields a stiffness ratio of $k_2/k_1 \approx 7.5 \times 10^8$, creating a temporal scale separation where the fast autocatalytic reaction R2 operates nearly a billion times faster than the slow initiation reaction R1.

  1194→\subsection{Comparison with Krylov-FSP-SSA}
  1195→\label{sec:comparison}
  1196→
  1197→We compare with the Krylov-FSP-SSA algorithm of Vo and Sidje~\cite{Sidje2015}, reviewed in detail by Dinh and Sidje~\cite{Dinh2016}. This baseline shares the same expand--evolve--prune skeleton and Krylov matrix exponential integrator but differs in three key respects: (i) pruning uses a joint probability-and-derivative threshold (Algorithm~2, Step~7 of~\cite{Vo2017}: state $i$ is dropped only if $p_i < \texttt{droptol}$ \emph{and} $(\mathbf{A}\boldsymbol{p})_i < \texttt{droptol}'$, without flux-based protection), (ii) time stepping uses a mass-conservation accept/reject criterion (halving $\tau$ when retained mass falls below $1 - \varepsilon_{\mathrm{FSP}}(t+\tau)/t_f$) rather than the flux-adaptive rule $\Delta t = \varepsilon_{\Delta t}/\Phi$, and (iii) expansion is SSA-driven with $r$-step reachability enumeration. Figure~\ref{fig:unified_comparison} and Table~\ref{tab:comparison_summary} compare the two methods on the three benchmark systems; each exposes a qualitatively distinct failure mode.
